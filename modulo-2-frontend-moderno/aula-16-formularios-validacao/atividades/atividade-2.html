Validação por Campo com Mensagens Acessíveis — Resposta

Aula 16 · Módulo 2: Frontend Moderno · ByteClass

O que esta atividade entrega

O formulário do Bicho Feliz com regras de validação numa função pura (validar), mensagens que só aparecem depois que a pessoa sai do campo, ligação entre campo e mensagem via aria-describedby e botão bloqueado enquanto houver pendência. O botão ainda não envia: o envio é a próxima prática.

src/utils/validar.js

// Início: src/utils/validar.js — regras de validação do agendamento
const TELEFONE = /^\(?\d{2}\)?\s?9?\d{4}-?\d{4}$/;

// Recebe os valores do formulário e devolve um objeto só com os campos inválidos
export function validar(valores) {
  const erros = {};
  if (valores.tutor.trim().length < 3) {
    erros.tutor = 'Informe o nome com pelo menos 3 letras.';
  }
  if (!TELEFONE.test(valores.telefone.trim())) {
    erros.telefone = 'Use o formato (32) 99999-0000.';
  }
  if (!valores.termos) {
    erros.termos = 'É preciso aceitar as regras para agendar.';
  }
  return erros;
}
// Fim de validar.js

src/components/CampoTexto.jsx

// Início: src/components/CampoTexto.jsx — campo com rótulo, erro e atributos ARIA
// Mostra o erro só quando o campo já foi visitado e liga a mensagem ao input
export default function CampoTexto({ id, rotulo, valor, erro, visivel, onChange, onBlur, ...resto }) {
  const mostrar = Boolean(visivel && erro);
  return (
    <div className="campo">
      <label htmlFor={id}>{rotulo}</label>
      <input id={id} name={id} value={valor} onChange={onChange} onBlur={onBlur}
        aria-invalid={mostrar} aria-describedby={mostrar ? `erro-${id}` : undefined} {...resto} />
      {mostrar && <span id={`erro-${id}`} className="erro">{erro}</span>}
    </div>
  );
}
// Fim de CampoTexto.jsx

src/components/FormAgendamento.jsx

// Início: src/components/FormAgendamento.jsx — validação por campo
import { useState } from 'react';
import CampoTexto from './CampoTexto.jsx';
import { validar } from '../utils/validar.js';

const inicial = { tutor: '', telefone: '', porte: 'pequeno', termos: false };

// Formulário que recalcula os erros a cada mudança e só os exibe nos campos visitados
export default function FormAgendamento() {
  const [form, setForm] = useState(inicial);
  const [tocados, setTocados] = useState({});
  const erros = validar(form);
  const pendentes = Object.keys(erros).length;

  // Atualiza o valor do campo alterado
  function atualizar(evento) {
    const { name, value, type, checked } = evento.target;
    setForm({ ...form, [name]: type === 'checkbox' ? checked : value });
  }

  // Marca o campo como visitado quando ele perde o foco
  function marcarTocado(evento) {
    setTocados({ ...tocados, [evento.target.name]: true });
  }

  return (
    <form className="card" noValidate>
      <h2>Agendar banho — Bicho Feliz</h2>
      <CampoTexto id="tutor" rotulo="Nome do tutor" valor={form.tutor} erro={erros.tutor}
        visivel={tocados.tutor} onChange={atualizar} onBlur={marcarTocado} />
      <CampoTexto id="telefone" rotulo="Telefone" type="tel" placeholder="(32) 99999-0000"
        valor={form.telefone} erro={erros.telefone} visivel={tocados.telefone}
        onChange={atualizar} onBlur={marcarTocado} />
      <div className="campo">
        <label htmlFor="porte">Porte do cachorro</label>
        <select id="porte" name="porte" value={form.porte} onChange={atualizar}>
          <option value="pequeno">Pequeno</option>
          <option value="medio">Médio</option>
          <option value="grande">Grande</option>
        </select>
      </div>
      <label className="check">
        <input type="checkbox" name="termos" checked={form.termos} onChange={atualizar} onBlur={marcarTocado}
          aria-invalid={Boolean(tocados.termos && erros.termos)}
          aria-describedby={tocados.termos && erros.termos ? 'erro-termos' : undefined} />
        Li e aceito as regras do pet shop
      </label>
      {tocados.termos && erros.termos && <p id="erro-termos" className="erro">{erros.termos}</p>}
      <button type="button" disabled={pendentes > 0}>Agendar</button>
      <p className="muted">{pendentes === 0 ? 'Tudo certo para agendar.' : `${pendentes} campo(s) pendente(s).`}</p>
    </form>
  );
}
// Fim de FormAgendamento.jsx

Como rodar

  1. No projeto da prática anterior, crie src/utils/validar.js e src/components/CampoTexto.jsx
  2. Substitua o FormAgendamento.jsx
  3. npm run dev, clique no nome e saia sem digitar
  4. Teste os telefones (32) 99999-0000, 32999990000 e 1234

O que observar

  • Os erros não são estado: são calculados a partir do formulário em cada renderização, então nunca ficam desatualizados.
  • Com o leitor de tela ligado, ao focar um campo inválido a mensagem é lida junto com o rótulo.
  • A validação no navegador melhora a experiência, mas não protege os dados: o servidor precisa validar de novo.
