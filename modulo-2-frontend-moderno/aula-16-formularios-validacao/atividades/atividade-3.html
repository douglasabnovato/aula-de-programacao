Envio com Estados, Proteção contra Duplo Clique e Retorno — Resposta

Aula 16 · Módulo 2: Frontend Moderno · ByteClass

O que esta atividade entrega

O envio completo do agendamento: onSubmit com preventDefault, validação de todos os campos no clique com foco no primeiro inválido, botão travado durante o envio, mensagem de sucesso ou falha numa região aria-live e formulário limpo após confirmar. O servidor é simulado com uma Promise; a chamada HTTP real é o assunto da Aula 17.

// Início: src/services/agendamentos.js — envio simulado (a API real entra na Aula 17)
// Simula uma chamada de rede de 1,5 s que falha quando o tutor se chama "erro"
export function enviarAgendamento(dados) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (dados.tutor.trim().toLowerCase() === 'erro') {
        reject(new Error('Servidor indisponível. Tente de novo em instantes.'));
      } else {
        resolve({ protocolo: Math.floor(Math.random() * 9000) + 1000 });
      }
    }, 1500);
  });
}

src/components/FormAgendamento.jsx

// Início: src/components/FormAgendamento.jsx — envio com estados e retorno ao usuário
import { useState } from 'react';
import CampoTexto from './CampoTexto.jsx';
import { validar } from '../utils/validar.js';
import { enviarAgendamento } from '../services/agendamentos.js';

const inicial = { tutor: '', telefone: '', porte: 'pequeno', termos: false };

// Formulário que valida no envio, bloqueia cliques repetidos e anuncia o resultado
export default function FormAgendamento() {
  const [form, setForm] = useState(inicial);
  const [tocados, setTocados] = useState({});
  const [status, setStatus] = useState('parado');
  const [mensagem, setMensagem] = useState('');
  const erros = validar(form);
  const enviando = status === 'enviando';

  // Atualiza o valor do campo alterado
  function atualizar(evento) {
    const { name, value, type, checked } = evento.target;
    setForm({ ...form, [name]: type === 'checkbox' ? checked : value });
  }

  // Marca o campo como visitado quando ele perde o foco
  function marcarTocado(evento) {
    setTocados({ ...tocados, [evento.target.name]: true });
  }

  // Valida, envia uma única vez e registra sucesso ou falha
  async function enviar(evento) {
    evento.preventDefault();
    if (enviando) return;
    const invalidos = Object.keys(erros);
    if (invalidos.length > 0) {
      setTocados({ tutor: true, telefone: true, termos: true });
      evento.target.elements[invalidos[0]].focus();
      return;
    }
    setStatus('enviando');
    setMensagem('');
    try {
      const resposta = await enviarAgendamento(form);
      setStatus('sucesso');
      setMensagem(`Agendamento confirmado. Protocolo ${resposta.protocolo}.`);
      setForm(inicial);
      setTocados({});
    } catch (falha) {
      setStatus('erro');
      setMensagem(falha.message);
    }
  }

  return (
    <form className="card" noValidate onSubmit={enviar}>
      <h2>Agendar banho — Bicho Feliz</h2>
      <CampoTexto id="tutor" rotulo="Nome do tutor" valor={form.tutor} erro={erros.tutor}
        visivel={tocados.tutor} onChange={atualizar} onBlur={marcarTocado} disabled={enviando} />
      <CampoTexto id="telefone" rotulo="Telefone" type="tel" placeholder="(32) 99999-0000"
        valor={form.telefone} erro={erros.telefone} visivel={tocados.telefone}
        onChange={atualizar} onBlur={marcarTocado} disabled={enviando} />
      <label className="check">
        <input type="checkbox" name="termos" checked={form.termos} onChange={atualizar} onBlur={marcarTocado}
          disabled={enviando} aria-invalid={Boolean(tocados.termos && erros.termos)}
          aria-describedby={tocados.termos && erros.termos ? 'erro-termos' : undefined} />
        Li e aceito as regras do pet shop
      </label>
      {tocados.termos && erros.termos && <p id="erro-termos" className="erro">{erros.termos}</p>}
      <button type="submit" disabled={enviando}>{enviando ? 'Enviando...' : 'Agendar'}</button>
      <p role="status" aria-live="polite" className={status === 'erro' ? 'erro' : 'sucesso'}>{mensagem}</p>
      <p className="muted">Dica de teste: use o nome "erro" para simular uma falha do servidor.</p>
    </form>
  );
}
// Fim de FormAgendamento.jsx

Como rodar

  1. Mantenha validar.js e CampoTexto.jsx da prática anterior
  2. Crie src/services/agendamentos.js e substitua o FormAgendamento.jsx
  3. npm run dev, clique em Agendar com tudo vazio e veja o foco ir para o nome
  4. Preencha corretamente, clique várias vezes seguidas; depois repita com o nome erro

O que observar

  • Sem preventDefault, o navegador recarrega a página no envio e todo o estado se perde.
  • Durante o envio o botão fica desabilitado e a função também sai cedo se já estiver enviando: duas barreiras contra registro duplicado.
  • role="status" com aria-live="polite" faz o leitor de tela anunciar o resultado sem roubar o foco.
