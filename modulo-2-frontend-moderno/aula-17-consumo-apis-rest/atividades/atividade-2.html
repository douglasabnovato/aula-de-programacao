CRUD de Tarefas contra uma API REST — Resposta

Aula 17 · Módulo 2: Frontend Moderno · ByteClass

O que esta atividade entrega

O painel da Entrega Já criando tarefas com POST, marcando como feitas com PATCH e excluindo com DELETE, sempre atualizando a lista a partir do estado anterior e bloqueando só o item que está sendo salvo. A API de teste responde como uma API real, mas não grava nada: por isso as tarefas criadas recebem um id local e não voltam a ser enviadas.

// Início: src/components/GerenciadorTarefas.jsx — POST, PATCH e DELETE numa API REST
import { useEffect, useState } from 'react';

const API = 'https://jsonplaceholder.typicode.com/todos';
const JSON_HEADERS = { 'Content-Type': 'application/json' };

// Lista de tarefas com criação, alternância de status e exclusão pela API
export default function GerenciadorTarefas() {
  const [tarefas, setTarefas] = useState([]);
  const [titulo, setTitulo] = useState('');
  const [ocupado, setOcupado] = useState(null);
  const [erro, setErro] = useState(null);

  useEffect(() => {
    // Carrega as cinco primeiras tarefas do usuário 1
    async function carregar() {
      try {
        const resposta = await fetch(`${API}?userId=1&_limit=5`);
        if (!resposta.ok) throw new Error(`Falha ${resposta.status} ao carregar.`);
        setTarefas(await resposta.json());
      } catch (falha) {
        setErro(falha.message);
      }
    }
    carregar();
  }, []);

  // Cria a tarefa com POST e coloca a resposta no topo da lista
  async function criar(evento) {
    evento.preventDefault();
    if (!titulo.trim()) return;
    setOcupado('nova');
    setErro(null);
    try {
      const resposta = await fetch(API, {
        method: 'POST',
        headers: JSON_HEADERS,
        body: JSON.stringify({ title: titulo.trim(), completed: false, userId: 1 }),
      });
      if (!resposta.ok) throw new Error(`Falha ${resposta.status} ao criar.`);
      const criada = await resposta.json();
      setTarefas((atuais) => [{ ...criada, id: Date.now(), local: true }, ...atuais]);
      setTitulo('');
    } catch (falha) {
      setErro(falha.message);
    } finally {
      setOcupado(null);
    }
  }

  // Alterna concluída com PATCH; tarefas criadas nesta sessão não existem na API de teste
  async function alternar(tarefa) {
    setOcupado(tarefa.id);
    setErro(null);
    try {
      if (!tarefa.local) {
        const resposta = await fetch(`${API}/${tarefa.id}`, {
          method: 'PATCH',
          headers: JSON_HEADERS,
          body: JSON.stringify({ completed: !tarefa.completed }),
        });
        if (!resposta.ok) throw new Error(`Falha ${resposta.status} ao atualizar.`);
      }
      setTarefas((atuais) => atuais.map((t) => (t.id === tarefa.id ? { ...t, completed: !t.completed } : t)));
    } catch (falha) {
      setErro(falha.message);
    } finally {
      setOcupado(null);
    }
  }

  // Exclui com DELETE e remove o item da lista
  async function excluir(tarefa) {
    setOcupado(tarefa.id);
    setErro(null);
    try {
      if (!tarefa.local) {
        const resposta = await fetch(`${API}/${tarefa.id}`, { method: 'DELETE' });
        if (!resposta.ok) throw new Error(`Falha ${resposta.status} ao excluir.`);
      }
      setTarefas((atuais) => atuais.filter((t) => t.id !== tarefa.id));
    } catch (falha) {
      setErro(falha.message);
    } finally {
      setOcupado(null);
    }
  }

  return (
    <section className="card">
      <h2>Gerenciar tarefas — Entrega Já</h2>
      <form className="nova" onSubmit={criar}>
        <input value={titulo} onChange={(e) => setTitulo(e.target.value)} placeholder="Nova tarefa" aria-label="Nova tarefa" />
        <button type="submit" disabled={ocupado === 'nova'}>{ocupado === 'nova' ? 'Salvando...' : 'Adicionar'}</button>
      </form>
      {erro && <p className="erro">{erro}</p>}
      <ul className="tarefas">
        {tarefas.map((t) => (
          <li key={t.id} className={t.completed ? 'feita' : ''}>
            <input type="checkbox" checked={t.completed} disabled={ocupado === t.id}
              onChange={() => alternar(t)} aria-label={`Concluir ${t.title}`} />
            <span>{t.title}</span>
            <button className="mini secundario" disabled={ocupado === t.id} onClick={() => excluir(t)}>Excluir</button>
          </li>
        ))}
      </ul>
    </section>
  );
}
// Fim de GerenciadorTarefas.jsx

Como rodar

  1. No projeto da prática anterior, crie src/components/GerenciadorTarefas.jsx
  2. Renderize <GerenciadorTarefas /> no App.jsx
  3. npm run dev com a aba Rede aberta e o filtro Fetch/XHR ligado
  4. Crie, marque e exclua tarefas conferindo o método e o status de cada requisição

O que observar

  • Sem o cabeçalho Content-Type: application/json, a API não entende o corpo e devolve só o id.
  • A atualização por função (setTarefas((atuais) => ...)) evita perder mudanças quando duas respostas chegam em sequência.
  • Os status esperados são 201 no POST e 200 no PATCH e no DELETE.
