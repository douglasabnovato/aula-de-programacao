Carrinho de Compras com Reducer e Contexto — Resposta

Aula 18 · Módulo 2: Frontend Moderno · ByteClass

O que esta atividade entrega

O carrinho da Tear Mineiro com todas as regras de alteração num único carrinhoReducer. O botão Comprar, o contador do cabeçalho e o painel do carrinho apenas disparam ações como { type: 'adicionado' }, e quantidade e total são calculados a partir dos itens, sem estado próprio.

src/contexts/CarrinhoContext.jsx

// Início: src/contexts/CarrinhoContext.jsx — carrinho com reducer e contexto
import { createContext, useContext, useReducer } from 'react';

const CarrinhoContext = createContext(null);

// Aplica cada ação ao carrinho devolvendo sempre um array novo
export function carrinhoReducer(itens, acao) {
  switch (acao.type) {
    case 'adicionado': {
      const existe = itens.find((i) => i.id === acao.produto.id);
      if (existe) {
        return itens.map((i) => (i.id === acao.produto.id ? { ...i, quantidade: i.quantidade + 1 } : i));
      }
      return [...itens, { ...acao.produto, quantidade: 1 }];
    }
    case 'quantidade_alterada':
      return itens
        .map((i) => (i.id === acao.id ? { ...i, quantidade: i.quantidade + acao.delta } : i))
        .filter((i) => i.quantidade > 0);
    case 'removido':
      return itens.filter((i) => i.id !== acao.id);
    case 'esvaziado':
      return [];
    default:
      throw new Error(`Ação desconhecida: ${acao.type}`);
  }
}

// Mantém o estado do carrinho e entrega itens e dispatch para a árvore
export function CarrinhoProvider({ children }) {
  const [itens, dispatch] = useReducer(carrinhoReducer, []);
  return <CarrinhoContext.Provider value={{ itens, dispatch }}>{children}</CarrinhoContext.Provider>;
}

// Lê o carrinho e calcula quantidade e total a partir dos itens
export function useCarrinho() {
  const contexto = useContext(CarrinhoContext);
  if (!contexto) {
    throw new Error('useCarrinho precisa ser usado dentro de <CarrinhoProvider>.');
  }
  const quantidade = contexto.itens.reduce((soma, i) => soma + i.quantidade, 0);
  const total = contexto.itens.reduce((soma, i) => soma + i.preco * i.quantidade, 0);
  return { ...contexto, quantidade, total };
}
// Fim de CarrinhoContext.jsx

src/components/Loja.jsx

// Início: src/components/Loja.jsx — componentes que só disparam ações
import { useCarrinho } from '../contexts/CarrinhoContext.jsx';
import { produtos, brl } from '../data/produtos.js';

// Contador do cabeçalho, sempre em dia com o carrinho
function Cabecalho() {
  const { quantidade } = useCarrinho();
  return (
    <header>
      <h2>Tear Mineiro</h2>
      <span>Carrinho: {quantidade} {quantidade === 1 ? 'item' : 'itens'}</span>
    </header>
  );
}

// Vitrine com o botão Comprar de cada produto
function Vitrine() {
  const { dispatch } = useCarrinho();
  return (
    <section className="vitrine">
      {produtos.map((p) => (
        <article key={p.id} className="produto">
          <strong>{p.nome}</strong>
          <span>{brl(p.preco)}</span>
          <button onClick={() => dispatch({ type: 'adicionado', produto: p })}>Comprar</button>
        </article>
      ))}
    </section>
  );
}

// Resumo do carrinho com ajuste de quantidade, remoção e total
function Carrinho() {
  const { itens, total, dispatch } = useCarrinho();
  if (itens.length === 0) {
    return <p className="carrinho">Seu carrinho está vazio.</p>;
  }
  return (
    <section className="carrinho">
      <ul>
        {itens.map((i) => (
          <li key={i.id}>
            <span>{i.nome}</span>
            <button className="secundario" onClick={() => dispatch({ type: 'quantidade_alterada', id: i.id, delta: -1 })}>−</button>
            <span>{i.quantidade}</span>
            <button className="secundario" onClick={() => dispatch({ type: 'quantidade_alterada', id: i.id, delta: 1 })}>+</button>
            <button className="secundario" onClick={() => dispatch({ type: 'removido', id: i.id })}>Remover</button>
          </li>
        ))}
      </ul>
      <p><strong>Total: {brl(total)}</strong></p>
      <button onClick={() => dispatch({ type: 'esvaziado' })}>Esvaziar carrinho</button>
    </section>
  );
}

// Página da loja reunindo as três partes
export default function Loja() {
  return (
    <div className="loja claro">
      <Cabecalho />
      <Vitrine />
      <Carrinho />
    </div>
  );
}
// Fim de Loja.jsx

src/App.jsx

// Início: src/App.jsx — provedor do carrinho na raiz
import { CarrinhoProvider } from './contexts/CarrinhoContext.jsx';
import Loja from './components/Loja.jsx';

// Componente raiz
export default function App() {
  return (
    <CarrinhoProvider>
      <Loja />
    </CarrinhoProvider>
  );
}
// Fim de App.jsx

Como rodar

  1. Mantenha src/data/produtos.js da prática anterior
  2. Crie src/contexts/CarrinhoContext.jsx e substitua Loja.jsx e App.jsx
  3. npm run dev, compre o mesmo produto duas vezes e confira a quantidade 2
  4. Use os botões −, + e Remover e confira o total a cada passo

O que observar

  • Com itens.push() e return itens, o React recebe o mesmo array e não renderiza de novo: o reducer precisa devolver um array novo.
  • O reducer é uma função pura: dá para testar com carrinhoReducer([], { type: 'esvaziado' }) no console, sem React.
  • Uma ação desconhecida lança erro na hora, em vez de passar em silêncio.
