Provedores Separados, Value Memorizado e Carrinho Persistido — Resposta

Aula 18 · Módulo 2: Frontend Moderno · ByteClass

O que esta atividade entrega

A Tear Mineiro com tema e carrinho em provedores separados, cada value memorizado com useMemo, a busca de produtos de volta ao estado local da vitrine e o carrinho salvo no localStorage. Os contadores de renderização na tela mostram quem é afetado por cada mudança.

src/contexts/carrinhoReducer.js

// Início: src/contexts/carrinhoReducer.js — regras do carrinho num arquivo próprio
// Aplica cada ação ao carrinho devolvendo sempre um array novo
export function carrinhoReducer(itens, acao) {
  switch (acao.type) {
    case 'adicionado': {
      const existe = itens.find((i) => i.id === acao.produto.id);
      if (existe) {
        return itens.map((i) => (i.id === acao.produto.id ? { ...i, quantidade: i.quantidade + 1 } : i));
      }
      return [...itens, { ...acao.produto, quantidade: 1 }];
    }
    case 'removido':
      return itens.filter((i) => i.id !== acao.id);
    case 'esvaziado':
      return [];
    default:
      throw new Error(`Ação desconhecida: ${acao.type}`);
  }
}
// Fim de carrinhoReducer.js

src/contexts/CarrinhoContext.jsx

// Início: src/contexts/CarrinhoContext.jsx — carrinho persistido e com value estável
import { createContext, useContext, useEffect, useMemo, useReducer } from 'react';
import { carrinhoReducer } from './carrinhoReducer.js';

const CarrinhoContext = createContext(null);
const CHAVE = 'tear-carrinho';

// Lê o carrinho salvo; se o storage estiver indisponível ou corrompido, começa vazio
function carregarInicial() {
  try {
    const salvo = localStorage.getItem(CHAVE);
    return salvo ? JSON.parse(salvo) : [];
  } catch {
    return [];
  }
}

// Provedor que inicia pelo storage, grava a cada mudança e memoriza o value
export function CarrinhoProvider({ children }) {
  const [itens, dispatch] = useReducer(carrinhoReducer, undefined, carregarInicial);

  useEffect(() => {
    try {
      localStorage.setItem(CHAVE, JSON.stringify(itens));
    } catch {
      console.warn('Não foi possível salvar o carrinho neste navegador.');
    }
  }, [itens]);

  const valor = useMemo(() => ({ itens, dispatch }), [itens]);
  return <CarrinhoContext.Provider value={valor}>{children}</CarrinhoContext.Provider>;
}

// Lê o carrinho e calcula a quantidade total de itens
export function useCarrinho() {
  const contexto = useContext(CarrinhoContext);
  if (!contexto) {
    throw new Error('useCarrinho precisa ser usado dentro de <CarrinhoProvider>.');
  }
  const quantidade = contexto.itens.reduce((soma, i) => soma + i.quantidade, 0);
  return { ...contexto, quantidade };
}
// Fim de CarrinhoContext.jsx

src/contexts/TemaContext.jsx

// Início: src/contexts/TemaContext.jsx — tema com value memorizado
import { createContext, useContext, useMemo, useState } from 'react';

const TemaContext = createContext(null);

// Provedor do tema; o value só muda quando o tema muda
export function TemaProvider({ children }) {
  const [tema, setTema] = useState('claro');
  const valor = useMemo(
    () => ({ tema, alternarTema: () => setTema((t) => (t === 'claro' ? 'escuro' : 'claro')) }),
    [tema]
  );
  return <TemaContext.Provider value={valor}>{children}</TemaContext.Provider>;
}

// Lê o tema com mensagem clara fora do provedor
export function useTema() {
  const contexto = useContext(TemaContext);
  if (!contexto) {
    throw new Error('useTema precisa ser usado dentro de <TemaProvider>.');
  }
  return contexto;
}
// Fim de TemaContext.jsx

src/App.jsx

// Início: src/App.jsx — provedores separados e compostos num só lugar
import { TemaProvider } from './contexts/TemaContext.jsx';
import { CarrinhoProvider } from './contexts/CarrinhoContext.jsx';
import Loja from './components/Loja.jsx';

// Reúne os provedores globais da aplicação, cada um com uma responsabilidade
function Provedores({ children }) {
  return (
    <TemaProvider>
      <CarrinhoProvider>{children}</CarrinhoProvider>
    </TemaProvider>
  );
}

// Componente raiz
export default function App() {
  return (
    <Provedores>
      <Loja />
    </Provedores>
  );
}
// Fim de App.jsx

src/components/Loja.jsx

// Início: src/components/Loja.jsx — busca local, contadores de renderização e contextos separados
import { memo, useRef, useState } from 'react';
import { useTema } from '../contexts/TemaContext.jsx';
import { useCarrinho } from '../contexts/CarrinhoContext.jsx';
import { produtos, brl } from '../data/produtos.js';

// Conta quantas vezes o componente renderizou, sem provocar nova renderização
function useRenders() {
  const renders = useRef(0);
  renders.current += 1;
  return renders.current;
}

// Cabeçalho: depende do tema e da quantidade do carrinho
function Cabecalho() {
  const { tema, alternarTema } = useTema();
  const { quantidade, dispatch } = useCarrinho();
  const renders = useRenders();
  return (
    <header>
      <h2>Tear Mineiro <span className="renders">(renders: {renders})</span></h2>
      <span>Carrinho: {quantidade}</span>
      <button className="secundario" onClick={() => dispatch({ type: 'esvaziado' })}>Esvaziar</button>
      <button onClick={alternarTema}>{tema === 'claro' ? 'Tema escuro' : 'Tema claro'}</button>
    </header>
  );
}

// Vitrine: a busca é estado local, porque só ela usa o texto digitado
function Vitrine() {
  const [busca, setBusca] = useState('');
  const { dispatch } = useCarrinho();
  const renders = useRenders();
  const visiveis = produtos.filter((p) => p.nome.toLowerCase().includes(busca.toLowerCase()));
  return (
    <section>
      <p className="renders">Vitrine renders: {renders}</p>
      <input value={busca} onChange={(e) => setBusca(e.target.value)} placeholder="Buscar produto" aria-label="Buscar produto" />
      <div className="vitrine" style={{ marginTop: 10 }}>
        {visiveis.map((p) => (
          <article key={p.id} className="produto">
            <strong>{p.nome}</strong>
            <span>{brl(p.preco)}</span>
            <button onClick={() => dispatch({ type: 'adicionado', produto: p })}>Comprar</button>
          </article>
        ))}
      </div>
    </section>
  );
}

// Rodapé memorizado: não lê nenhum contexto, então não renderiza quando eles mudam
const Rodape = memo(function Rodape() {
  const renders = useRenders();
  return <p className="renders">Rodapé renders: {renders} · Tear Mineiro, artesanato de Minas.</p>;
});

// Moldura da loja com a classe do tema
export default function Loja() {
  const { tema } = useTema();
  return (
    <div className={`loja ${tema}`}>
      <Cabecalho />
      <Vitrine />
      <Rodape />
    </div>
  );
}
// Fim de Loja.jsx

Como rodar

  1. Mantenha src/data/produtos.js e salve os cinco arquivos acima
  2. npm run dev e digite na busca observando os contadores
  3. Compre dois produtos e recarregue a página: o carrinho continua lá
  4. Na aba Application do DevTools, veja a chave tear-carrinho no Local Storage

O que observar

  • Digitar na busca renderiza só a Vitrine; se a busca estivesse num contexto global, o Cabeçalho também contaria.
  • O Rodape com memo não lê contexto e não recebe props, então não renderiza quando o tema ou o carrinho mudam.
  • O terceiro argumento do useReducer lê o storage uma vez só, na montagem; o try/catch protege contra navegador sem storage ou JSON corrompido.
