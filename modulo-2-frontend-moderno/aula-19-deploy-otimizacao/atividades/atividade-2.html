Medindo e Otimizando uma Lista de 2.000 Sessões — Resposta

Aula 19 · Módulo 2: Frontend Moderno · ByteClass

O que esta atividade entrega

A programação do Cine Clube Paraibuna com 2.000 sessões, um filtro que custa 80 ms de propósito e três ajustes medidos: useMemo para o filtro rodar só quando a busca muda, memo na lista e useCallback para a função de favoritar não quebrar o memo. Os contadores na tela mostram quantas vezes o filtro e a lista rodaram.

src/data/sessoes.js

// Início: src/data/sessoes.js — gerador de sessões fictícias e filtro caro de propósito
const FILMES = ['Noite no Paraibuna', 'O Trem das Onze', 'Serra Azul', 'Café com Neblina', 'A Última Sessão', 'Rua Halfeld, 1950'];
const SALAS = ['Sala 1', 'Sala 2', 'Sala Rosa'];

// Gera uma lista de sessões com título, sala e horário
export function gerarSessoes(total) {
  return Array.from({ length: total }, (_, i) => ({
    id: i + 1,
    titulo: `${FILMES[i % FILMES.length]} #${Math.floor(i / FILMES.length) + 1}`,
    sala: SALAS[i % SALAS.length],
    horario: `${String(14 + (i % 8)).padStart(2, '0')}:${i % 2 ? '30' : '00'}`,
  }));
}

// Segura a execução pelo tempo indicado, simulando um cálculo pesado
function esperar(ms) {
  const fim = performance.now() + ms;
  let agora = performance.now();
  while (agora < fim) {
    agora = performance.now();
  }
}

// Filtra as sessões pelo texto, sem diferenciar acentos nem maiúsculas
export function filtrarSessoes(sessoes, busca) {
  esperar(80);
  const termo = busca.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
  return sessoes.filter((s) => s.titulo.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().includes(termo));
}
// Fim de sessoes.js

src/components/ListaSessoes.jsx

// Início: src/components/ListaSessoes.jsx — lista memorizada
import { memo, useRef } from 'react';

// Lista que só renderiza de novo quando sessões, favoritos ou onFavoritar mudam
function ListaSessoes({ sessoes, favoritos, onFavoritar }) {
  const renders = useRef(0);
  renders.current += 1;
  return (
    <>
      <p className="metricas">Lista renderizada {renders.current} vez(es) · mostrando {Math.min(sessoes.length, 100)} de {sessoes.length}</p>
      <ul className="sessoes">
        {sessoes.slice(0, 100).map((s) => (
          <li key={s.id}>
            <span>{s.horario} · {s.titulo} · {s.sala}</span>
            <button className="secundario" onClick={() => onFavoritar(s.id)}>
              {favoritos.includes(s.id) ? '★' : '☆'}
            </button>
          </li>
        ))}
      </ul>
    </>
  );
}

export default memo(ListaSessoes);
// Fim de ListaSessoes.jsx

src/App.jsx

// Início: src/App.jsx — busca com useMemo e callback estável
import { useCallback, useMemo, useRef, useState } from 'react';
import ListaSessoes from './components/ListaSessoes.jsx';
import { gerarSessoes, filtrarSessoes } from './data/sessoes.js';

const SESSOES = gerarSessoes(2000);

// Página de programação: o filtro caro só roda quando a busca muda
export default function App() {
  const [busca, setBusca] = useState('');
  const [claro, setClaro] = useState(false);
  const [favoritos, setFavoritos] = useState([]);
  const execucoes = useRef(0);

  const visiveis = useMemo(() => {
    execucoes.current += 1;
    return filtrarSessoes(SESSOES, busca);
  }, [busca]);

  const favoritar = useCallback((id) => {
    setFavoritos((atuais) => (atuais.includes(id) ? atuais.filter((f) => f !== id) : [...atuais, id]));
  }, []);

  return (
    <div className={claro ? 'claro-19' : ''} style={{ padding: 12, borderRadius: 12 }}>
      <section className="card">
        <h2>Programação — Cine Clube Paraibuna</h2>
        <div className="linha">
          <input value={busca} onChange={(e) => setBusca(e.target.value)} placeholder="Buscar filme" aria-label="Buscar filme" />
          <button className="secundario" onClick={() => setClaro(!claro)}>Trocar tema</button>
        </div>
        <p className="metricas">Filtro executado {execucoes.current} vez(es) · favoritos: {favoritos.length}</p>
        <ListaSessoes sessoes={visiveis} favoritos={favoritos} onFavoritar={favoritar} />
      </section>
    </div>
  );
}
// Fim de App.jsx

Como rodar

  1. Crie os três arquivos num projeto Vite + React
  2. npm run dev e instale a extensão React Developer Tools
  3. Na aba Profiler, grave enquanto digita e enquanto clica em Trocar tema
  4. Compare as durações antes e depois de cada otimização

O que observar

  • Trocar o tema não aumenta o contador do filtro nem o da lista: nenhuma das dependências mudou.
  • Com onFavoritar={(id) => ...} escrito direto no JSX, cada renderização cria uma função nova e o memo da lista deixa de funcionar.
  • Mostrar só 100 itens com slice reduz o DOM; otimização sem medição antes costuma só deixar o código mais difícil de ler.
