Chamada Resiliente entre Serviços — Código de Referência

Aula 35 · Módulo 4: Fullstack, DevOps e Projeto Final · ByteClass

O que esta atividade entrega

Um cliente HTTP com tempo máximo de espera, nova tentativa com intervalo crescente e disjuntor, usado pelo serviço de pedidos para continuar respondendo quando notificações está lento ou fora do ar.

pedidos/src/clienteNotificacoes.js

// Início: clienteNotificacoes.js — chamada ao serviço de notificações com timeout, retry e disjuntor
const URL_BASE = process.env.NOTIFICACOES_URL || 'http://notificacoes:4001';
const TENTATIVAS = 3;
const LIMITE_FALHAS = 3;
const TEMPO_ABERTO_MS = 15000;

const disjuntor = { estado: 'fechado', falhas: 0, abertoEm: 0 };

// espera o intervalo informado em milissegundos
function esperar(ms) {
  return new Promise((resolver) => setTimeout(resolver, ms));
}

// decide se o disjuntor deixa a chamada passar
function podeChamar() {
  if (disjuntor.estado !== 'aberto') return true;
  if (Date.now() - disjuntor.abertoEm >= TEMPO_ABERTO_MS) {
    disjuntor.estado = 'meio-aberto';
    return true;
  }
  return false;
}

// zera as falhas e fecha o disjuntor
function registrarSucesso() {
  disjuntor.estado = 'fechado';
  disjuntor.falhas = 0;
}

// conta a falha e abre o disjuntor ao atingir o limite
function registrarFalha() {
  disjuntor.falhas += 1;
  if (disjuntor.estado === 'meio-aberto' || disjuntor.falhas >= LIMITE_FALHAS) {
    disjuntor.estado = 'aberto';
    disjuntor.abertoEm = Date.now();
  }
}

// faz uma única chamada com limite de dois segundos
async function chamarUmaVez(corpo) {
  const resposta = await fetch(`${URL_BASE}/notificacoes`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(corpo),
    signal: AbortSignal.timeout(2000)
  });
  if (resposta.status >= 500) {
    throw new Error(`notificações respondeu ${resposta.status}`);
  }
  return { status: resposta.status, corpo: await resposta.json() };
}

// tenta notificar até três vezes e devolve se conseguiu
async function notificarPedido(corpo) {
  if (!podeChamar()) {
    return { enviada: false, motivo: 'disjuntor aberto' };
  }
  for (let tentativa = 1; tentativa <= TENTATIVAS; tentativa++) {
    try {
      const resultado = await chamarUmaVez(corpo);
      registrarSucesso();
      return { enviada: resultado.status === 201, resultado };
    } catch (erro) {
      if (tentativa === TENTATIVAS) {
        registrarFalha();
        return { enviada: false, motivo: erro.message };
      }
      await esperar(200 * 2 ** (tentativa - 1));
    }
  }
}

module.exports = { notificarPedido, disjuntor };
// Fim de clienteNotificacoes.js

pedidos/server.js

// Início: server.js do serviço de pedidos — pedido aceito mesmo com notificações fora do ar
const express = require('express');
const { notificarPedido, disjuntor } = require('./src/clienteNotificacoes');

const app = express();
app.use(express.json());

const pedidos = [];

// cria o pedido e tenta notificar sem deixar a falha vazar para o cliente
app.post('/pedidos', async (req, res) => {
  const { cliente, itens } = req.body;
  if (!cliente || !Array.isArray(itens) || itens.length === 0) {
    return res.status(400).json({ erro: 'cliente e itens são obrigatórios' });
  }
  const pedido = { id: pedidos.length + 1, cliente, itens, notificacao: 'pendente' };
  pedidos.push(pedido);

  const envio = await notificarPedido({ pedidoId: pedido.id, cliente });
  pedido.notificacao = envio.enviada ? 'enviada' : 'pendente';
  if (!envio.enviada) console.warn(`Pedido ${pedido.id}: notificação pendente (${envio.motivo})`);
  res.status(201).json(pedido);
});

// mostra o estado do disjuntor para quem opera o sistema
app.get('/saude/dependencias', (req, res) => {
  res.json({ notificacoes: { ...disjuntor } });
});

const porta = process.env.PORT || 3000;
app.listen(porta, () => console.log(`Pedidos ouvindo na porta ${porta}`));
// Fim de server.js

notificacoes/atraso.js

// Início: atraso.js — middleware do serviço de notificações que simula lentidão pelo ambiente
const ATRASO_MS = Number(process.env.ATRASO_MS || 0);

// segura a requisição pelo tempo configurado antes de seguir
function simularAtraso(req, res, next) {
  if (ATRASO_MS > 0) return setTimeout(next, ATRASO_MS);
  next();
}

module.exports = simularAtraso;
// Fim de atraso.js

Como rodar

  1. Registre app.use(require('./atraso')) no server.js de notificações, antes das rotas
  2. Suba com ATRASO_MS=30000 docker compose up --build e crie um pedido: a resposta chega em cerca de 7 segundos (3 tentativas de 2s mais as esperas)
  3. Crie mais dois pedidos e consulte http://localhost:3000/saude/dependencias: o disjuntor está aberto
  4. Crie outro pedido: a resposta agora é imediata, com notificacao: "pendente"
  5. Rode docker compose up -d sem o atraso, espere 15 segundos e crie um pedido: o disjuntor volta a fechado

O que observar

  • Um status 400 de notificações não entra no retry: erro de dado enviado não melhora repetindo.
  • Com o disjuntor aberto, pedidos nem tenta a chamada — isso poupa o serviço doente de receber mais carga.
  • Os pedidos com notificação pendente precisam de um destino: numa versão real, eles iriam para uma fila e seriam reenviados depois.
