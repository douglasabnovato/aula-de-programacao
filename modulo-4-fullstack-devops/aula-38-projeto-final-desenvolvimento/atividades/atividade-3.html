api/test/agendamentos.test.js

// Início: agendamentos.test.js — critérios de aceite da história 3 como testes
const { test, before, after } = require('node:test');
const assert = require('node:assert');
const app = require('../src/app');
const { gerarTokenDeTeste } = require('./apoio');

let servidor;
let base;
let token;

// sobe a API numa porta livre antes dos testes
before(async () => {
  servidor = app.listen(0);
  base = `http://localhost:${servidor.address().port}`;
  token = gerarTokenDeTeste();
});

// derruba a API ao final
after(() => servidor.close());

// monta a requisição de criação com o token
function criar(corpo, comToken = true) {
  return fetch(`${base}/agendamentos`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(comToken ? { Authorization: `Bearer ${token}` } : {})
    },
    body: JSON.stringify(corpo)
  });
}

const horario = {
  profissionalId: 1,
  clienteNome: 'Cliente Teste',
  inicio: '2026-12-01T14:00:00Z',
  fim: '2026-12-01T15:00:00Z'
};

// dado um horário livre, quando crio, então recebo 201
test('cria agendamento em horário livre', async () => {
  const resposta = await criar(horario);
  assert.strictEqual(resposta.status, 201);
});

// dado um horário ocupado, quando crio outro sobreposto, então recebo 409
test('recusa agendamento sobreposto', async () => {
  const resposta = await criar({ ...horario, inicio: '2026-12-01T14:30:00Z', fim: '2026-12-01T15:30:00Z' });
  assert.strictEqual(resposta.status, 409);
  const corpo = await resposta.json();
  assert.strictEqual(corpo.erro, 'Horário indisponível');
});

// sem token, a rota protegida responde 401
test('exige autenticação', async () => {
  const resposta = await criar(horario, false);
  assert.strictEqual(resposta.status, 401);
});
// Fim de agendamentos.test.js

Roteiro do revisor

  1. Leia a descrição e a issue antes do código: você sabe o que deveria mudar?
  2. Rode os passos de Como testar na sua máquina.
  3. Confira se os testes cobrem o critério de aceite e o erro principal.
  4. Procure regra de negócio só no frontend ou só na API quando deveria estar nos dois lados.
  5. Deixe ao menos um comentário em forma de pergunta ("o que acontece se...").
  6. Marque comentários opcionais com o prefixo Sugestão: para o autor saber o que não bloqueia.
  7. Aprove quando a mudança deixa o código melhor do que estava, mesmo sem ser perfeita.

O que observar

  • O teste usa app.listen(0): o sistema escolhe uma porta livre e a suíte não conflita com a API rodando.
  • Os testes dependem do banco de teste limpo; a esteira sobe um PostgreSQL só para isso.
  • Os nomes dos testes repetem o Dado, Quando, Então do escopo: quem lê o relatório da esteira entende o que quebrou.
