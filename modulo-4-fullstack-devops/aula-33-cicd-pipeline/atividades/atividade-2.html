test/tarefas.test.js — com o runner nativo do Node

const { test } = require('node:test');
const assert = require('node:assert');
const servico = require('../src/services/tarefas.service');

test('recusa tarefa sem título', () => {
  assert.throws(() => servico.criar({ titulo: '   ' }));
});

test('cria tarefa com título válido', () => {
  const tarefa = servico.criar({ titulo: 'Estudar CI' });
  assert.strictEqual(tarefa.titulo, 'Estudar CI');
  assert.strictEqual(tarefa.feita, false);
});

test('gera ids diferentes para tarefas diferentes', () => {
  const a = servico.criar({ titulo: 'Uma' });
  const b = servico.criar({ titulo: 'Outra' });
  assert.notStrictEqual(a.id, b.id);
});

Sem instalar nada: node --test já vem no Node 20.

Workflow com jobs em paralelo

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20', cache: 'npm' }
      - run: npm ci
      - run: npm run lint

  teste:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20', cache: 'npm' }
      - run: npm ci
      - run: npm test
      - run: npm audit --audit-level=high

Por que separar em jobs

  • Rodam em paralelo: o retorno chega mais rápido.
  • O resultado fica granular: você vê se foi o lint ou o teste que quebrou, sem abrir o log.
  • Cada um pode ter cache e ambiente próprios.

O teste que vale a pena escrever primeiro

Não é o do caminho feliz — esse você testa à mão o tempo todo. É o da regra de negócio na borda: título vazio, prazo no passado, id que não existe. São os que quebram em produção e ninguém percebe em desenvolvimento.
