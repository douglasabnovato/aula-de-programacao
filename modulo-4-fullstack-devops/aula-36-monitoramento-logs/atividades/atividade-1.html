Logs Estruturados com Pino — Código de Referência

Aula 36 · Módulo 4: Fullstack, DevOps e Projeto Final · ByteClass

O que esta atividade entrega

Uma API Express com log em JSON para toda requisição, identificador propagado no cabeçalho X-Request-Id, nível ajustado pelo status e campos sensíveis ocultados.

src/logger.js

// Início: logger.js — logger JSON único da aplicação, com campos sensíveis ocultados
const pino = require('pino');

const logger = pino({
  level: process.env.LOG_LEVEL || 'info',
  base: {
    servico: 'api-agendamentos',
    ambiente: process.env.NODE_ENV || 'development'
  },
  redact: {
    paths: ['req.headers.authorization', 'req.headers.cookie', '*.senha', '*.token'],
    censor: '[OCULTO]'
  }
});

module.exports = logger;
// Fim de logger.js

app.js

// Início: app.js — API de agendamentos com log por requisição e identificador rastreável
const express = require('express');
const pinoHttp = require('pino-http');
const { randomUUID } = require('node:crypto');
const logger = require('./src/logger');

const app = express();
app.use(express.json());

app.use(pinoHttp({
  logger,
  genReqId: (req, res) => {
    const id = req.headers['x-request-id'] || randomUUID();
    res.setHeader('X-Request-Id', id);
    return id;
  },
  customLogLevel: (req, res, erro) => {
    if (erro || res.statusCode >= 500) return 'error';
    if (res.statusCode >= 400) return 'warn';
    return 'info';
  }
}));

const agendamentos = [];

// cria um agendamento e registra o evento com contexto
app.post('/agendamentos', (req, res) => {
  const { pet, horario } = req.body;
  if (!pet || !horario) {
    req.log.warn({ campos: Object.keys(req.body) }, 'agendamento recusado por falta de dados');
    return res.status(400).json({ erro: 'pet e horario são obrigatórios' });
  }
  const agendamento = { id: agendamentos.length + 1, pet, horario };
  agendamentos.push(agendamento);
  req.log.info({ agendamentoId: agendamento.id }, 'agendamento criado');
  res.status(201).json(agendamento);
});

// recebe o login e mostra que a senha não aparece no log
app.post('/login', (req, res) => {
  req.log.info({ corpo: req.body }, 'login recebido');
  res.status(204).end();
});

// registra a falha com a pilha e responde sem expor detalhes
app.use((erro, req, res, next) => {
  req.log.error({ err: erro }, 'falha não tratada');
  res.status(500).json({ erro: 'Erro interno', requestId: req.id });
});

const porta = process.env.PORT || 3000;
app.listen(porta, () => logger.info({ porta }, 'API no ar'));
// Fim de app.js

Como rodar

  1. npm install express pino pino-http
  2. node app.js | npx pino-pretty para leitura em desenvolvimento, ou só node app.js para ver o JSON cru
  3. curl -X POST localhost:3000/agendamentos -H "Content-Type: application/json" -H "x-request-id: teste-123" -d '{"pet":"Paçoca","horario":"14:00"}'
  4. curl -X POST localhost:3000/login -H "Content-Type: application/json" -d '{"email":"ana@exemplo.com","senha":"123456"}'

O que observar

  • As linhas de agendamento criado e da requisição trazem o mesmo reqId, que também volta no cabeçalho da resposta.
  • No login, o campo senha aparece como [OCULTO]: a ocultação acontece antes de a linha ser escrita.
  • O erro devolve o requestId ao cliente: com ele, o suporte acha a linha exata do log sem expor a pilha.
