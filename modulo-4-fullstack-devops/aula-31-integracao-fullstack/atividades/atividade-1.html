CONTRATO.md — o documento que evita retrabalho

## GET /tarefas
Resposta 200: [{ "id": 1, "titulo": "Estudar", "feita": false }]

## GET /tarefas/:id
Resposta 200: { "id": 1, "titulo": "Estudar", "feita": false }
Resposta 404: { "erro": "Tarefa não encontrada" }

## POST /tarefas
Corpo:       { "titulo": "Nova tarefa" }
Resposta 201: { "id": 3, "titulo": "Nova tarefa", "feita": false }
Resposta 400: { "erro": "O campo titulo é obrigatório" }

## DELETE /tarefas/:id
Resposta 204: sem corpo
Resposta 404: { "erro": "Tarefa não encontrada" }

src/api/cliente.js

const BASE = import.meta.env.VITE_API_URL;

async function requisitar(caminho, opcoes = {}) {
  const resposta = await fetch(`${BASE}${caminho}`, {
    headers: { 'Content-Type': 'application/json' },
    ...opcoes
  });

  // fetch nao rejeita em 4xx nem 5xx. Sem esta checagem, erro vira sucesso.
  if (!resposta.ok) {
    const corpo = await resposta.json().catch(() => ({}));
    throw new Error(corpo.erro || `Falha ${resposta.status}`);
  }

  return resposta.status === 204 ? null : resposta.json();
}

export const listarTarefas = () => requisitar('/tarefas');
export const criarTarefa = (titulo) =>
  requisitar('/tarefas', { method: 'POST', body: JSON.stringify({ titulo }) });
export const removerTarefa = (id) =>
  requisitar(`/tarefas/${id}`, { method: 'DELETE' });

A armadilha do fetch

O fetch só rejeita a promessa quando a requisição não sai — rede caída, DNS errado. Um 404 ou um 500 chegam como resposta normal. Sem if (!resposta.ok), o catch nunca dispara e o erro passa como se fosse dado.
