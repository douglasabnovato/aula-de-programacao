ListaTarefas.jsx

import { useEffect, useState } from 'react';
import { listarTarefas } from '../api/cliente';

export function ListaTarefas() {
  const [dados, setDados] = useState([]);
  const [carregando, setCarregando] = useState(true);
  const [erro, setErro] = useState(null);

  useEffect(() => {
    const controlador = new AbortController();

    async function buscar() {
      try {
        setCarregando(true);
        setErro(null);
        const lista = await listarTarefas({ signal: controlador.signal });
        setDados(lista);
      } catch (e) {
        if (e.name !== 'AbortError') setErro(e.message);
      } finally {
        setCarregando(false);
      }
    }

    buscar();
    return () => controlador.abort();
  }, []);

  if (carregando) return <p role="status">Carregando tarefas…</p>;

  if (erro) return (
    <div role="alert">
      <p>Não foi possível carregar: {erro}</p>
      <button onClick={() => window.location.reload()}>Tentar de novo</button>
    </div>
  );

  if (dados.length === 0) return (
    <p>Nenhuma tarefa ainda. Crie a primeira para começar.</p>
  );

  return (
    <ul>
      {dados.map(t => <li key={t.id}>{t.titulo}</li>)}
    </ul>
  );
}

Os quatro cenários para testar

  1. Sucesso com itens — a lista aparece.
  2. Sucesso sem itens — apague tudo no banco; deve aparecer o convite, não erro.
  3. API fora do ar — pare o servidor; deve aparecer a mensagem com botão.
  4. Rede lenta — no DevTools, escolha Slow 3G e veja o estado de carregando durar.

Estado vazio não é erro. Confundir os dois é o defeito mais comum em tela de listagem: o usuário acha que o sistema quebrou quando na verdade ele ainda não cadastrou nada.
