Token JWT: Emissão e Middleware — Código de Referência

Aula 22 · Módulo 3: Backend e Banco de Dados · ByteClass

O ponto da atividade

O JWT é assinado, não escondido. Qualquer pessoa com o token lê o payload. A assinatura garante que ninguém o alterou — não que ninguém o leu.

Emissão, no login

const jwt = require('jsonwebtoken');

const token = jwt.sign(
  { sub: usuario.id, email: usuario.email },
  process.env.JWT_SECRET,
  { expiresIn: '1h' }
);

res.json({ token });

src/middlewares/auth.middleware.js

const jwt = require('jsonwebtoken');

module.exports = (req, res, proximo) => {
  const cabecalho = req.headers.authorization || '';
  const [esquema, token] = cabecalho.split(' ');

  if (esquema !== 'Bearer' || !token) {
    return res.status(401).json({ erro: 'Token ausente' });
  }

  try {
    const dados = jwt.verify(token, process.env.JWT_SECRET);
    req.usuarioId = dados.sub;
    proximo();
  } catch (erro) {
    // Expirado e adulterado caem os dois aqui, e recebem a mesma resposta.
    return res.status(401).json({ erro: 'Token inválido ou expirado' });
  }
};

Usando em uma rota

const autenticar = require('../middlewares/auth.middleware');

rotas.get('/tarefas', autenticar, controlador.listar);
