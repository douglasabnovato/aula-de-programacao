Erros de Domínio e Problem Details — Código de Referência

Aula 26 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Uma hierarquia de erros com status próprio, um serviço que lança esses erros em vez de montar resposta, e o middleware que traduz qualquer falha para application/problem+json.

src/erros/AppError.js

// Início: hierarquia de erros de domínio com o status HTTP de cada um
class AppError extends Error {
  // guarda status, título curto e detalhes opcionais
  constructor(status, titulo, detalhe, extras = {}) {
    super(detalhe);
    this.name = this.constructor.name;
    this.status = status;
    this.titulo = titulo;
    this.extras = extras;
  }
}

class NaoEncontrado extends AppError {
  // recurso pedido não existe
  constructor(recurso, id) {
    super(404, 'Recurso não encontrado', `${recurso} ${id} não existe`);
  }
}

class RegraViolada extends AppError {
  // pedido bem formado que quebra uma regra de negócio
  constructor(detalhe) {
    super(422, 'Regra de negócio violada', detalhe);
  }
}

class ValidacaoFalhou extends AppError {
  // corpo da requisição fora do formato esperado
  constructor(problemas) {
    super(400, 'Dados inválidos', 'Um ou mais campos não passaram na validação', { problemas });
  }
}

module.exports = { AppError, NaoEncontrado, RegraViolada, ValidacaoFalhou };
// Fim de AppError.js

src/services/tarefas.service.js

// Início: serviço de tarefas que lança erros de domínio em vez de montar respostas HTTP
const { NaoEncontrado, RegraViolada } = require('../erros/AppError');

// cria o serviço sobre o repositório recebido
function criarServicoTarefas(repositorio) {
  // busca a tarefa ou lança 404
  async function detalhar(id) {
    const tarefa = await repositorio.buscarPorId(id);
    if (!tarefa) throw new NaoEncontrado('Tarefa', id);
    return tarefa;
  }

  // marca como feita, recusando tarefa já concluída
  async function concluir(id) {
    const tarefa = await detalhar(id);
    if (tarefa.feita) throw new RegraViolada('A tarefa já está concluída');
    return repositorio.atualizar(id, { feita: true });
  }

  return { detalhar, concluir };
}

module.exports = criarServicoTarefas;
// Fim de tarefas.service.js

src/middlewares/erro.middleware.js

// Início: tradução de qualquer erro para o formato Problem Details (RFC 9457)
const { AppError, NaoEncontrado } = require('../erros/AppError');

// último handler de rota: caminho que ninguém atendeu vira 404
function rotaInexistente(req, res, proximo) {
  proximo(new NaoEncontrado('Rota', req.originalUrl));
}

// middleware de erro: quatro argumentos, registrado por último
function tratarErro(erro, req, res, proximo) {
  const conhecido = erro instanceof AppError;
  const status = conhecido ? erro.status : 500;

  console.error(JSON.stringify({
    nivel: status >= 500 ? 'error' : 'warn',
    id: req.id,
    status,
    mensagem: erro.message,
    stack: status >= 500 ? erro.stack : undefined
  }));

  res.status(status).type('application/problem+json').json({
    type: 'about:blank',
    title: conhecido ? erro.titulo : 'Erro interno',
    status,
    detail: conhecido ? erro.message : 'Algo falhou no servidor. Informe o id abaixo ao suporte.',
    instance: req.originalUrl,
    requestId: req.id,
    ...(conhecido ? erro.extras : {})
  });
}

module.exports = { rotaInexistente, tratarErro };
// Fim de erro.middleware.js

src/app.js

// Início: app com rotas assíncronas, handler de 404 e erro centralizado
const express = require('express');
const { requestId, logRequisicao } = require('./middlewares');
const { rotaInexistente, tratarErro } = require('./middlewares/erro.middleware');
const criarServicoTarefas = require('./services/tarefas.service');
const TarefaSupabaseRepository = require('./infra/repositories/TarefaSupabaseRepository');

const app = express();
const servico = criarServicoTarefas(new TarefaSupabaseRepository());

app.use(requestId);
app.use(logRequisicao);
app.use(express.json({ limit: '10kb' }));

// detalhe sem try/catch: o Express 5 encaminha a rejeição ao tratarErro
app.get('/tarefas/:id', async (req, res) => {
  res.json(await servico.detalhar(req.params.id));
});

// conclusão que pode gerar 404 ou 422
app.patch('/tarefas/:id/concluir', async (req, res) => {
  res.json(await servico.concluir(req.params.id));
});

app.use(rotaInexistente);
app.use(tratarErro);

module.exports = app;
// Fim de app.js

Como rodar

  1. Confirme o Express 5 com npm ls express
  2. Acrescente ao repositório o método atualizar(id, campos) usando update do supabase-js
  3. node server.js
  4. Teste GET /tarefas/9999, PATCH /tarefas/1/concluir duas vezes e GET /qualquer-coisa

O que observar

  • Nenhuma rota tem try/catch: no Express 5, a Promise rejeitada vai sozinha para o tratarErro.
  • O erro 500 devolve ao cliente só uma frase e o requestId; o stack fica no log, onde o suporte procura pelo id.
  • O serviço não conhece res: ele lança NaoEncontrado e quem decide o formato HTTP é o middleware.
