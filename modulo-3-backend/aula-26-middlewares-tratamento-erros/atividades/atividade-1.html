Cadeia de Middlewares Globais — Código de Referência

Aula 26 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Dois middlewares globais — identificador por requisição e log estruturado com duração — exportados de um ponto só e registrados na ordem em que a requisição passa.

src/middlewares/requestId.js

// Início: middleware que dá um identificador único a cada requisição
const crypto = require('crypto');

const FORMATO_VALIDO = /^[A-Za-z0-9-]{8,64}$/;

// reaproveita o id recebido quando é seguro ou gera um novo
function requestId(req, res, proximo) {
  const recebido = req.get('X-Request-Id');
  req.id = recebido && FORMATO_VALIDO.test(recebido) ? recebido : crypto.randomUUID();
  res.set('X-Request-Id', req.id);
  proximo();
}

module.exports = requestId;
// Fim de requestId.js

src/middlewares/logRequisicao.js

// Início: middleware que mede a duração e registra cada requisição em uma linha JSON

// registra método, rota, status e duração quando a resposta termina
function logRequisicao(req, res, proximo) {
  const inicio = process.hrtime.bigint();
  res.on('finish', () => {
    const duracaoMs = Number(process.hrtime.bigint() - inicio) / 1e6;
    console.log(JSON.stringify({
      nivel: 'info',
      id: req.id,
      metodo: req.method,
      rota: req.originalUrl,
      status: res.statusCode,
      duracaoMs: Number(duracaoMs.toFixed(1)),
      em: new Date().toISOString()
    }));
  });
  proximo();
}

module.exports = logRequisicao;
// Fim de logRequisicao.js

src/middlewares/index.js

// Início: ponto único de exportação dos middlewares globais
const requestId = require('./requestId');
const logRequisicao = require('./logRequisicao');

module.exports = { requestId, logRequisicao };
// Fim de index.js

src/app.js

// Início: app com a cadeia de middlewares registrada na ordem em que a requisição passa
const express = require('express');
const { requestId, logRequisicao } = require('./middlewares');
const rotasTarefas = require('./routes/tarefas.routes');

const app = express();

app.use(requestId);
app.use(logRequisicao);
app.use(express.json({ limit: '10kb' }));

app.use('/tarefas', rotasTarefas);

module.exports = app;
// Fim de app.js

Como rodar

  1. Salve os arquivos nas pastas indicadas
  2. node server.js
  3. Faça uma chamada com curl -i http://localhost:3000/tarefas
  4. Procure no terminal a linha JSON com o mesmo id do cabeçalho X-Request-Id

O que observar

  • O log é escrito no evento finish: é só ali que o status final e a duração são conhecidos.
  • O id recebido do cliente passa por uma expressão regular antes de ser aceito — valor de cabeçalho é entrada do usuário.
  • Uma linha JSON por requisição permite filtrar o log por id, status ou rota com qualquer ferramenta.
