Validação com Zod — Código de Referência

Aula 26 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Esquemas de entrada declarados num arquivo só, um middleware genérico que valida qualquer rota e respostas 400 que apontam o campo exato, no mesmo formato Problem Details.

src/esquemas/tarefa.esquema.js

// Início: esquemas de entrada das rotas de tarefas
const { z } = require('zod');

const criarTarefaEsquema = z.strictObject({
  titulo: z.string({ error: 'titulo deve ser texto' })
    .trim()
    .min(3, { error: 'titulo precisa de pelo menos 3 caracteres' })
    .max(120, { error: 'titulo aceita no máximo 120 caracteres' }),
  prioridade: z.number()
    .int({ error: 'prioridade deve ser inteiro' })
    .min(1, { error: 'prioridade vai de 1 a 5' })
    .max(5, { error: 'prioridade vai de 1 a 5' })
    .optional(),
  prazo: z.coerce.date({ error: 'prazo deve ser uma data válida' }).optional()
});

const atualizarTarefaEsquema = criarTarefaEsquema.partial();

module.exports = { criarTarefaEsquema, atualizarTarefaEsquema };
// Fim de tarefa.esquema.js

src/middlewares/validar.js

// Início: middleware genérico que valida o corpo contra um esquema Zod
const { ValidacaoFalhou } = require('../erros/AppError');

// devolve um middleware que recusa o corpo fora do esquema
function validar(esquema) {
  return function (req, res, proximo) {
    const resultado = esquema.safeParse(req.body ?? {});
    if (!resultado.success) {
      const problemas = resultado.error.issues.map((issue) => ({
        campo: issue.path.join('.') || '(corpo)',
        mensagem: issue.message
      }));
      return proximo(new ValidacaoFalhou(problemas));
    }
    req.body = resultado.data;
    proximo();
  };
}

module.exports = validar;
// Fim de validar.js

src/routes/tarefas.routes.js

// Início: rotas de tarefas com validação declarada antes do controlador
const express = require('express');
const validar = require('../middlewares/validar');
const { criarTarefaEsquema, atualizarTarefaEsquema } = require('../esquemas/tarefa.esquema');
const controlador = require('../controllers/tarefas.controller');

const router = express.Router();

router.get('/', controlador.listar);
router.get('/:id', controlador.detalhar);
router.post('/', validar(criarTarefaEsquema), controlador.criar);
router.patch('/:id', validar(atualizarTarefaEsquema), controlador.atualizar);

module.exports = router;
// Fim de tarefas.routes.js

Como rodar

  1. npm install zod
  2. Salve os arquivos e registre o roteador com app.use('/tarefas', rotasTarefas)
  3. node server.js
  4. Envie { "titulo": "a", "prioridade": 99, "prazo": "amanhã", "admin": true } para POST /tarefas

O que observar

  • Um único corpo inválido devolve todos os problemas de uma vez, cada um com seu campo — o cliente corrige tudo numa tentativa.
  • strictObject recusa campos que não estão no esquema; sem ele, admin: true passaria calado até o banco.
  • req.body = resultado.data entrega ao controlador o dado já limpo: trim aplicado e prazo convertido em Date.
