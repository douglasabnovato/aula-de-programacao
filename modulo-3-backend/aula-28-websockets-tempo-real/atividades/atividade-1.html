WebSocket com ws — Código de Referência

Aula 28 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Um servidor WebSocket em Node que recebe pedidos em JSON e distribui a todos os painéis abertos, e um cliente HTML que usa o WebSocket nativo do navegador.

servidor-ws.js

// Início: servidor WebSocket puro que distribui pedidos a todos os painéis conectados
const { WebSocketServer, WebSocket } = require('ws');

const wss = new WebSocketServer({ port: 8080 });

// envia uma mensagem JSON para todas as conexões abertas
function difundir(mensagem) {
  const texto = JSON.stringify(mensagem);
  wss.clients.forEach((cliente) => {
    if (cliente.readyState === WebSocket.OPEN) cliente.send(texto);
  });
}

// trata cada nova conexão
wss.on('connection', (socket) => {
  socket.send(JSON.stringify({ tipo: 'boas-vindas', dados: { conectados: wss.clients.size } }));

  // interpreta a mensagem recebida e repassa pedidos a todos
  socket.on('message', (bruto) => {
    let mensagem;
    try {
      mensagem = JSON.parse(bruto.toString());
    } catch {
      return socket.send(JSON.stringify({ tipo: 'erro', dados: 'Mensagem precisa ser JSON' }));
    }
    if (mensagem.tipo === 'pedido') difundir({ tipo: 'pedido-novo', dados: mensagem.dados });
  });

  // informa quantos continuam conectados
  socket.on('close', () => console.log(`Clientes conectados: ${wss.clients.size}`));
});

console.log('WebSocket em ws://localhost:8080');
// Fim de servidor-ws.js

cliente.html

<!-- Início: painel mínimo que recebe pedidos pelo WebSocket nativo do navegador -->
<!DOCTYPE html>
<html lang="pt-BR">
<head>
  <meta charset="UTF-8">
  <title>Painel de pedidos</title>
</head>
<body>
  <h1>Pedidos</h1>
  <button id="enviar">Simular pedido</button>
  <ul id="lista"></ul>
  <script>
    // Início: conexão com o servidor e renderização das mensagens
    const socket = new WebSocket('ws://localhost:8080');
    const lista = document.getElementById('lista');

    // escreve uma linha no painel
    function registrar(texto) {
      const item = document.createElement('li');
      item.textContent = texto;
      lista.prepend(item);
    }

    // trata cada mensagem vinda do servidor
    socket.onmessage = (evento) => {
      const mensagem = JSON.parse(evento.data);
      registrar(`${mensagem.tipo}: ${JSON.stringify(mensagem.dados)}`);
    };

    // envia um pedido de exemplo
    document.getElementById('enviar').onclick = () => {
      socket.send(JSON.stringify({ tipo: 'pedido', dados: { item: 'X-Salada', mesa: 4 } }));
    };
    // Fim do script do painel
  </script>
</body>
</html>
<!-- Fim de cliente.html -->

Como rodar

  1. npm init -y e npm install ws
  2. node servidor-ws.js
  3. Abra cliente.html em três abas
  4. Clique em Simular pedido em uma delas

O que observar

  • Na aba Rede, a conexão aparece uma vez só, com status 101; as mensagens passam por ela sem novas requisições.
  • O try/catch no JSON.parse impede que uma mensagem malformada derrube o processo para todos.
  • Não há salas nem reconexão: tudo que o Socket.IO oferece na próxima atividade teria de ser escrito à mão aqui.
