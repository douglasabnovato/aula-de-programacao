Socket.IO Autenticado — Código de Referência

Aula 28 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Middleware que verifica o JWT no handshake, autorização de entrada em sala pelas unidades do token, validação do payload e limite de eventos por conexão.

src/socket-auth.js

// Início: autenticação no handshake, autorização por sala e limite de eventos por conexão
const jwt = require('jsonwebtoken');

const JANELA_MS = 10000;
const MAX_EVENTOS = 20;

// verifica o JWT enviado em auth e guarda o usuário na conexão
function autenticar(socket, proximo) {
  const token = socket.handshake.auth?.token;
  if (!token) return proximo(new Error('nao autorizado: token ausente'));
  try {
    socket.data.usuario = jwt.verify(token, process.env.JWT_SECRET);
    proximo();
  } catch {
    proximo(new Error('nao autorizado: token inválido ou expirado'));
  }
}

// conta eventos por janela e derruba a conexão que exagerar
function limitarEventos(socket) {
  let inicioJanela = Date.now();
  let contagem = 0;
  socket.use((pacote, proximo) => {
    const agora = Date.now();
    if (agora - inicioJanela > JANELA_MS) {
      inicioJanela = agora;
      contagem = 0;
    }
    contagem += 1;
    if (contagem > MAX_EVENTOS) return socket.disconnect(true);
    proximo();
  });
}

// registra os eventos já com as regras de acesso
function registrarEventos(io) {
  io.use(autenticar);

  io.on('connection', (socket) => {
    limitarEventos(socket);

    // só entra na sala se a unidade constar no token
    socket.on('entrar-unidade', (unidade, confirmar = () => {}) => {
      if (typeof unidade !== 'string') return confirmar({ ok: false, erro: 'unidade inválida' });
      const permitidas = socket.data.usuario.unidades || [];
      if (!permitidas.includes(unidade)) return confirmar({ ok: false, erro: 'sem acesso a esta unidade' });
      socket.join(unidade);
      confirmar({ ok: true });
    });
  });
}

module.exports = registrarEventos;
// Fim de socket-auth.js

scripts/gerar-tokens.js

// Início: gera tokens de teste para duas cozinhas diferentes
require('dotenv').config();
const jwt = require('jsonwebtoken');

// assina um token com as unidades permitidas
function gerar(nome, unidades) {
  return jwt.sign({ sub: nome, unidades }, process.env.JWT_SECRET, { expiresIn: '1h' });
}

console.log('Cozinha Centro:', gerar('cozinha-centro', ['centro']));
console.log('Cozinha São Pedro:', gerar('cozinha-sao-pedro', ['sao-pedro']));
// Fim de gerar-tokens.js

public/painel-auth.js

// Início: trecho do painel que conecta com token e trata a recusa
const token = sessionStorage.getItem('token');
const socket = io({ auth: { token }, reconnectionAttempts: 5 });

// mostra o motivo quando o servidor recusa a conexão
socket.on('connect_error', (erro) => {
  document.getElementById('estado').textContent = `Conexão recusada: ${erro.message}`;
});

// pede entrada na sala e informa se foi negada
socket.on('connect', () => {
  socket.emit('entrar-unidade', 'centro', (resposta) => {
    if (!resposta.ok) document.getElementById('estado').textContent = resposta.erro;
  });
});
// Fim do trecho do painel

Como rodar

  1. npm install jsonwebtoken dotenv e defina JWT_SECRET no .env
  2. No server.js, troque o io.on('connection') da atividade anterior por require('./src/socket-auth')(io)
  3. node scripts/gerar-tokens.js e salve um token no sessionStorage do painel
  4. Teste sem token, com o token de São Pedro e com o token do Centro

O que observar

  • A recusa acontece antes de connection: quem não tem token nunca chega a registrar evento nenhum.
  • O token vai em auth, não na URL — URL aparece em log de servidor e de proxy.
  • O limite derruba só a conexão abusiva; as outras cozinhas continuam recebendo normalmente.
