Painel com Socket.IO — Código de Referência

Aula 28 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Um servidor Express e Socket.IO na mesma porta, onde o pedido criado pela API REST chega só ao painel da unidade certa, com confirmação de recebimento e reentrada automática na sala após reconectar.

server.js

// Início: servidor Express + Socket.IO na mesma porta, com pedidos separados por unidade
const http = require('http');
const path = require('path');
const express = require('express');
const { Server } = require('socket.io');

const app = express();
const servidor = http.createServer(app);
const io = new Server(servidor);

const pedidos = [];
let proximoId = 1;

app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

// cria o pedido pela API REST e avisa só a cozinha da unidade
app.post('/pedidos', (req, res) => {
  const { unidade, item } = req.body;
  if (!unidade || !item) return res.status(400).json({ erro: 'Informe unidade e item' });
  const pedido = { id: proximoId++, unidade, item, status: 'novo' };
  pedidos.push(pedido);
  io.to(unidade).emit('pedido-novo', pedido);
  res.status(201).json(pedido);
});

// trata cada painel conectado
io.on('connection', (socket) => {
  // coloca o painel na sala da unidade
  socket.on('entrar-unidade', (unidade) => {
    socket.join(unidade);
    socket.emit('entrou', unidade);
  });

  // marca o pedido como pronto e confirma ao painel pelo callback
  socket.on('pedido-pronto', (id, confirmar) => {
    const pedido = pedidos.find((p) => p.id === Number(id));
    if (!pedido) return confirmar({ ok: false, erro: 'Pedido não encontrado' });
    pedido.status = 'pronto';
    io.to(pedido.unidade).emit('pedido-atualizado', pedido);
    confirmar({ ok: true });
  });
});

servidor.listen(3000, () => console.log('Painel em http://localhost:3000/painel.html'));
// Fim de server.js

public/painel.html

<!-- Início: painel da cozinha que entra na sala da unidade e reentra ao reconectar -->
<!DOCTYPE html>
<html lang="pt-BR">
<head>
  <meta charset="UTF-8">
  <title>Cozinha</title>
</head>
<body>
  <label>Unidade
    <select id="unidade">
      <option value="centro">Centro</option>
      <option value="sao-pedro">São Pedro</option>
    </select>
  </label>
  <p id="estado">Conectando...</p>
  <ul id="lista"></ul>
  <script src="/socket.io/socket.io.js"></script>
  <script>
    // Início: conexão Socket.IO do painel
    const socket = io();
    const seletor = document.getElementById('unidade');
    const lista = document.getElementById('lista');
    const estado = document.getElementById('estado');

    // entra (ou reentra) na sala escolhida
    function entrar() {
      socket.emit('entrar-unidade', seletor.value);
    }

    // desenha um pedido com botão de pronto
    function desenhar(pedido) {
      const item = document.createElement('li');
      item.textContent = `#${pedido.id} ${pedido.item} — ${pedido.status} `;
      const botao = document.createElement('button');
      botao.textContent = 'Pronto';
      botao.onclick = () => socket.emit('pedido-pronto', pedido.id, (resposta) => {
        if (!resposta.ok) alert(resposta.erro);
      });
      item.append(botao);
      lista.prepend(item);
    }

    socket.on('connect', () => { estado.textContent = 'Conectado'; entrar(); });
    socket.on('disconnect', () => { estado.textContent = 'Sem conexão — tentando de novo...'; });
    socket.on('pedido-novo', desenhar);
    socket.on('pedido-atualizado', desenhar);
    seletor.onchange = () => location.reload();
    // Fim do script do painel
  </script>
</body>
</html>
<!-- Fim de painel.html -->

Como rodar

  1. npm install express socket.io
  2. node server.js
  3. Abra http://localhost:3000/painel.html em duas abas, uma para cada unidade
  4. Envie POST /pedidos com { "unidade": "centro", "item": "Pastel" }

O que observar

  • O emit vem da rota REST: quem cria o pedido não precisa estar conectado por socket.
  • Salas vivem no servidor e somem quando a conexão cai; por isso o painel reentra dentro do evento connect.
  • O callback de pedido-pronto é a confirmação de que o servidor processou — sem ele, o painel não sabe se a mensagem chegou.
