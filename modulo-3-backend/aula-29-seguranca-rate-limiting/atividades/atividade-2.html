Rate Limiting em Camadas — Código de Referência

Aula 29 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Quatro limitadores com propósitos diferentes — global, rota cara, por usuário e login — registrados na ordem certa, com resposta 429 em Problem Details e um script para medir o comportamento.

src/middlewares/limites.js

// Início: limites de requisição em camadas — global, rota cara, por usuário e login
const { rateLimit, ipKeyGenerator } = require('express-rate-limit');

// resposta 429 no formato Problem Details com o tempo de espera
function responderLimite(req, res, proximo, opcoes) {
  const segundos = Math.ceil(opcoes.windowMs / 1000);
  res.status(opcoes.statusCode).type('application/problem+json').json({
    type: 'about:blank',
    title: 'Limite de requisições atingido',
    status: opcoes.statusCode,
    detail: `Tente novamente em até ${segundos} segundos.`,
    instance: req.originalUrl,
    requestId: req.id
  });
}

const base = { standardHeaders: 'draft-8', legacyHeaders: false, handler: responderLimite };

const limiteGlobal = rateLimit({ ...base, windowMs: 15 * 60 * 1000, limit: 300 });

const limiteRelatorios = rateLimit({ ...base, windowMs: 60 * 1000, limit: 20 });

const limitePorUsuario = rateLimit({
  ...base,
  windowMs: 60 * 1000,
  limit: 60,
  // usa o usuário autenticado como chave e o IP como reserva
  keyGenerator(req) {
    return req.usuarioId ? `usuario:${req.usuarioId}` : ipKeyGenerator(req.ip);
  }
});

const limiteLogin = rateLimit({
  ...base,
  windowMs: 15 * 60 * 1000,
  limit: 5,
  skipSuccessfulRequests: true
});

module.exports = { limiteGlobal, limiteRelatorios, limitePorUsuario, limiteLogin };
// Fim de limites.js

src/app.js

// Início: onde cada camada de limite entra no app
const express = require('express');
const helmet = require('helmet');
const corsConfigurado = require('./config/cors');
const autenticar = require('./middlewares/auth.middleware');
const { limiteGlobal, limiteRelatorios, limitePorUsuario, limiteLogin } = require('./middlewares/limites');
const rotasAuth = require('./routes/auth.routes');
const rotasTarefas = require('./routes/tarefas.routes');
const rotasRelatorios = require('./routes/relatorios.routes');

const app = express();

app.set('trust proxy', 1);
app.use(helmet());
app.use(corsConfigurado);
app.use(limiteGlobal);
app.use(express.json({ limit: '10kb' }));

app.use('/auth/login', limiteLogin);
app.use('/auth', rotasAuth);

app.use(autenticar);
app.use(limitePorUsuario);
app.use('/tarefas', rotasTarefas);
app.use('/relatorios', limiteRelatorios, rotasRelatorios);

module.exports = app;
// Fim de app.js

scripts/rajada.js

// Início: dispara uma rajada de chamadas e conta os status recebidos
const URL_ALVO = process.env.URL_ALVO || 'http://localhost:3000/relatorios';
const TOKEN = process.env.TOKEN_TESTE || '';

// executa as chamadas em sequência e imprime o resumo
async function rajada(total = 30) {
  const contagem = {};
  for (let i = 0; i < total; i += 1) {
    const resposta = await fetch(URL_ALVO, { headers: { Authorization: `Bearer ${TOKEN}` } });
    contagem[resposta.status] = (contagem[resposta.status] || 0) + 1;
    if (resposta.status === 429 && !contagem.primeiro429) contagem.primeiro429 = i + 1;
  }
  console.log(contagem);
}

rajada();
// Fim de rajada.js

Como rodar

  1. npm install express-rate-limit (versão 8 ou superior, que exporta ipKeyGenerator)
  2. node server.js
  3. TOKEN_TESTE=... node scripts/rajada.js
  4. Compare a contagem de status com os limites configurados

O que observar

  • limitePorUsuario vem depois de autenticar: antes dele, req.usuarioId ainda não existe.
  • trust proxy faz o req.ip ser o do cliente quando a API roda atrás do proxy da hospedagem; sem isso, todos parecem o mesmo IP.
  • O store padrão guarda a contagem na memória do processo; com mais de uma instância, cada uma conta separado.
