CORS com Credenciais — Código de Referência

Aula 29 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

A configuração de CORS com lista de origens vinda do ambiente, credenciais, métodos e cabeçalhos explícitos e cache de preflight, mais o cookie de sessão e a chamada do front que exercita tudo isso.

src/config/cors.js

// Início: CORS com lista de origens por ambiente, credenciais e cache de preflight
const cors = require('cors');

const ORIGENS = (process.env.ORIGENS_PERMITIDAS || '')
  .split(',')
  .map((o) => o.trim())
  .filter(Boolean);

// libera só origens da lista; requisições sem Origin (curl, servidor) seguem sem cabeçalhos CORS
function verificarOrigem(origem, callback) {
  if (!origem) return callback(null, false);
  callback(null, ORIGENS.includes(origem));
}

const corsConfigurado = cors({
  origin: verificarOrigem,
  credentials: true,
  methods: ['GET', 'POST', 'PATCH', 'DELETE'],
  allowedHeaders: ['Content-Type', 'Authorization', 'X-Request-Id'],
  exposedHeaders: ['X-Request-Id', 'RateLimit', 'RateLimit-Policy'],
  maxAge: 600
});

module.exports = corsConfigurado;
// Fim de cors.js

src/auth/cookieSessao.js

// Início: emissão do cookie de sessão com atributos definidos pelo ambiente
const EM_PRODUCAO = process.env.NODE_ENV === 'production';

// grava o token num cookie que o JavaScript da página não consegue ler
function emitirCookieSessao(res, token) {
  res.cookie('sessao', token, {
    httpOnly: true,
    secure: EM_PRODUCAO,
    sameSite: EM_PRODUCAO ? 'none' : 'lax',
    maxAge: 60 * 60 * 1000,
    path: '/'
  });
}

module.exports = emitirCookieSessao;
// Fim de cookieSessao.js

public/teste-cors.js

// Início: chamada do front que dispara preflight e envia o cookie
// envia a atualização da tarefa com credenciais
async function concluirTarefa(id) {
  const resposta = await fetch(`http://localhost:3000/tarefas/${id}`, {
    method: 'PATCH',
    credentials: 'include',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ feita: true })
  });
  console.log(resposta.status, resposta.headers.get('X-Request-Id'));
}

concluirTarefa(1);
// Fim de teste-cors.js

Como rodar

  1. npm install cors e defina ORIGENS_PERMITIDAS no .env
  2. Registre app.use(require('./src/config/cors')) antes das rotas
  3. Sirva uma página na porta 5500 que carrega teste-cors.js
  4. Abra a aba Rede e compare a primeira e a segunda chamada

O que observar

  • A primeira chamada PATCH gera um OPTIONS; as seguintes, dentro de 600 segundos, reaproveitam a resposta em cache.
  • Origem fora da lista não recebe erro do servidor: recebe resposta sem Access-Control-Allow-Origin, e quem bloqueia é o navegador.
  • sameSite: 'none' exige secure; por isso os dois mudam juntos quando o ambiente é produção.
