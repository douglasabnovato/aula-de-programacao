Upload Seguro — Código de Referência

Aula 27 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Três barreiras em sequência: limite de tamanho e quantidade no Multer, filtro pelo tipo declarado e checagem da assinatura real nos primeiros bytes. Erros do Multer viram 413 ou 400 no formato Problem Details.

src/uploads/detectarImagem.js

// Início: detecção do tipo real de imagem pelos primeiros bytes do arquivo
const ASSINATURAS = [
  { tipo: 'image/jpeg', extensao: 'jpg', confere: (b) => b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff },
  { tipo: 'image/png', extensao: 'png', confere: (b) => b.subarray(0, 4).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47])) },
  { tipo: 'image/webp', extensao: 'webp', confere: (b) => b.toString('ascii', 0, 4) === 'RIFF' && b.toString('ascii', 8, 12) === 'WEBP' }
];

// devolve { tipo, extensao } da imagem ou null quando o conteúdo não é imagem aceita
function detectarImagem(buffer) {
  if (!Buffer.isBuffer(buffer) || buffer.length < 12) return null;
  const achada = ASSINATURAS.find((a) => a.confere(buffer));
  return achada ? { tipo: achada.tipo, extensao: achada.extensao } : null;
}

module.exports = detectarImagem;
// Fim de detectarImagem.js

src/uploads/upload.js

// Início: Multer em memória com limite de tamanho e filtro de tipo declarado
const multer = require('multer');
const { AppError } = require('../erros/AppError');

const TIPOS_ACEITOS = ['image/jpeg', 'image/png', 'image/webp'];

const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 2 * 1024 * 1024, files: 1 },
  // primeira barreira: recusa o que nem declara ser imagem aceita
  fileFilter(req, arquivo, callback) {
    if (TIPOS_ACEITOS.includes(arquivo.mimetype)) return callback(null, true);
    callback(new AppError(415, 'Tipo de arquivo não suportado', 'Envie JPEG, PNG ou WebP'));
  }
});

module.exports = upload;
// Fim de upload.js

src/middlewares/traduzirErroMulter.js

// Início: trecho do middleware de erro da aula 26 que traduz falhas do Multer
const multer = require('multer');
const { AppError } = require('../erros/AppError');

// converte MulterError em AppError com o status adequado
function traduzirErroMulter(erro) {
  if (!(erro instanceof multer.MulterError)) return erro;
  if (erro.code === 'LIMIT_FILE_SIZE') {
    return new AppError(413, 'Arquivo grande demais', 'O limite é de 2 MB por arquivo');
  }
  return new AppError(400, 'Upload inválido', `${erro.message} (campo: ${erro.field || 'desconhecido'})`);
}

module.exports = traduzirErroMulter;
// Fim de traduzirErroMulter.js — chame no início de tratarErro: erro = traduzirErroMulter(erro)

src/routes/anexos.routes.js

// Início: rota de anexo que só aceita imagem de verdade
const crypto = require('crypto');
const express = require('express');
const upload = require('../uploads/upload');
const detectarImagem = require('../uploads/detectarImagem');
const { AppError } = require('../erros/AppError');

const router = express.Router();

// valida presença, assinatura real e define o nome pelo servidor
router.post('/tarefas/:id/anexos', upload.single('foto'), (req, res) => {
  if (!req.file) throw new AppError(400, 'Arquivo ausente', 'Envie a imagem no campo foto');

  const detectado = detectarImagem(req.file.buffer);
  if (!detectado) throw new AppError(415, 'Conteúdo não é imagem', 'O arquivo não é JPEG, PNG ou WebP de verdade');

  const nomeFinal = `${crypto.randomUUID()}.${detectado.extensao}`;
  res.status(201).json({ nomeFinal, tipo: detectado.tipo, tamanho: req.file.size });
});

module.exports = router;
// Fim de anexos.routes.js

Como rodar

  1. Salve os arquivos e chame traduzirErroMulter na primeira linha do tratarErro da aula 26
  2. node server.js
  3. Envie uma imagem válida, uma de mais de 2 MB e um .txt renomeado para .jpg
  4. Compare os status 201, 413 e 415

O que observar

  • O fileFilter só lê o que o cliente declarou; é a assinatura em detectarImagem que prova o conteúdo.
  • Com memoryStorage, o arquivo recusado nunca toca o disco do servidor.
  • A extensão final vem do tipo detectado, então um .php disfarçado nunca é gravado com a extensão que escolheu.
