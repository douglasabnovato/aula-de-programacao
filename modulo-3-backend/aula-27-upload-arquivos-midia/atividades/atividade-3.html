Supabase Storage — Código de Referência

Aula 27 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

A foto validada vai para um bucket privado do Supabase, os metadados vão para a tabela anexos e a leitura acontece por link assinado que expira em cinco minutos.

db/anexos.sql

-- Início: tabela de metadados dos anexos guardados no Supabase Storage
create table anexos (
  id bigint generated always as identity primary key,
  tarefa_id bigint not null references tarefas(id) on delete cascade,
  caminho text not null unique,
  tipo text not null,
  tamanho integer not null check (tamanho > 0),
  criado_em timestamptz not null default now()
);

create index anexos_tarefa_id_idx on anexos (tarefa_id);
-- Fim de anexos.sql

src/services/anexos.service.js

// Início: serviço que guarda o arquivo no bucket privado e os metadados no banco
const crypto = require('crypto');
const supabase = require('../infra/db/supabase');
const { AppError, NaoEncontrado } = require('../erros/AppError');

const BUCKET = 'anexos';

// envia o arquivo, registra os metadados e desfaz o upload se o registro falhar
async function salvarAnexo(tarefaId, arquivo, detectado) {
  const caminho = `tarefas/${tarefaId}/${crypto.randomUUID()}.${detectado.extensao}`;

  const envio = await supabase.storage.from(BUCKET).upload(caminho, arquivo.buffer, {
    contentType: detectado.tipo,
    upsert: false
  });
  if (envio.error) throw new AppError(502, 'Falha no armazenamento', envio.error.message);

  const { data, error } = await supabase
    .from('anexos')
    .insert({ tarefa_id: tarefaId, caminho, tipo: detectado.tipo, tamanho: arquivo.size })
    .select('id, caminho, tipo, tamanho')
    .single();

  if (error) {
    await supabase.storage.from(BUCKET).remove([caminho]);
    throw error;
  }
  return data;
}

// gera um link temporário para o anexo informado
async function gerarLink(anexoId, segundos = 300) {
  const { data: anexo, error } = await supabase.from('anexos').select('caminho').eq('id', anexoId).maybeSingle();
  if (error) throw error;
  if (!anexo) throw new NaoEncontrado('Anexo', anexoId);

  const assinado = await supabase.storage.from(BUCKET).createSignedUrl(anexo.caminho, segundos);
  if (assinado.error) throw new AppError(502, 'Falha no armazenamento', assinado.error.message);
  return { url: assinado.data.signedUrl, expiraEmSegundos: segundos };
}

module.exports = { salvarAnexo, gerarLink };
// Fim de anexos.service.js

src/routes/anexos.routes.js

// Início: rotas de anexos ligadas ao Supabase Storage
const express = require('express');
const upload = require('../uploads/upload');
const detectarImagem = require('../uploads/detectarImagem');
const { salvarAnexo, gerarLink } = require('../services/anexos.service');
const { AppError } = require('../erros/AppError');

const router = express.Router();

// recebe, valida e guarda a imagem da tarefa
router.post('/tarefas/:id/anexos', upload.single('foto'), async (req, res) => {
  if (!req.file) throw new AppError(400, 'Arquivo ausente', 'Envie a imagem no campo foto');
  const detectado = detectarImagem(req.file.buffer);
  if (!detectado) throw new AppError(415, 'Conteúdo não é imagem', 'O arquivo não é JPEG, PNG ou WebP de verdade');
  res.status(201).json(await salvarAnexo(Number(req.params.id), req.file, detectado));
});

// devolve um link assinado de cinco minutos
router.get('/anexos/:id/link', async (req, res) => {
  res.json(await gerarLink(Number(req.params.id)));
});

module.exports = router;
// Fim de anexos.routes.js

Como rodar

  1. Crie o bucket privado anexos em Storage no painel do Supabase
  2. Execute db/anexos.sql no SQL Editor
  3. node server.js com SUPABASE_URL e SUPABASE_SECRET_KEY no .env
  4. Envie uma foto e depois chame GET /anexos/1/link

O que observar

  • O banco guarda o caminho, não o arquivo: trocar de provedor de armazenamento depois muda só o serviço.
  • Se o insert falhar, o remove apaga o arquivo recém-enviado — sem isso, o bucket acumula arquivos sem dono.
  • O link assinado vale para qualquer pessoa enquanto não expira; por isso o prazo curto.
