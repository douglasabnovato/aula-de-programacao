API de Tarefas em Express — Código de Referência

Aula 21 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

Uma API com quatro rotas REST sobre uma lista em memória, com status HTTP corretos e validação de entrada.

app.js

const express = require('express');

const app = express();
app.use(express.json());

let tarefas = [
  { id: 1, titulo: 'Estudar Event Loop', feita: false },
  { id: 2, titulo: 'Escrever a primeira rota', feita: true }
];

app.get('/tarefas', (req, res) => {
  res.json(tarefas);
});

app.get('/tarefas/:id', (req, res) => {
  const tarefa = tarefas.find(t => t.id === Number(req.params.id));
  if (!tarefa) return res.status(404).json({ erro: 'Tarefa não encontrada' });
  res.json(tarefa);
});

app.post('/tarefas', (req, res) => {
  const { titulo } = req.body;
  if (!titulo || titulo.trim() === '') {
    return res.status(400).json({ erro: 'O campo titulo é obrigatório' });
  }
  const nova = { id: tarefas.length + 1, titulo: titulo.trim(), feita: false };
  tarefas.push(nova);
  res.status(201).json(nova);
});

app.delete('/tarefas/:id', (req, res) => {
  const antes = tarefas.length;
  tarefas = tarefas.filter(t => t.id !== Number(req.params.id));
  if (tarefas.length === antes) return res.status(404).json({ erro: 'Tarefa não encontrada' });
  res.status(204).end();
});

app.listen(3000, () => console.log('API em http://localhost:3000'));

Tabela de status usada

  • 200 — listagem e detalhe devolvidos com sucesso
  • 201 — recurso criado; devolve o objeto novo
  • 204 — removido com sucesso, sem corpo de resposta
  • 400 — o cliente mandou dado inválido
  • 404 — o id pedido não existe

Ponto de atenção

tarefas.length + 1 como gerador de id quebra assim que alguém apaga um item: dois registros podem receber o mesmo número. É proposital — na aula 23 esse id sai da mão do controlador e vai para a camada de dados.
