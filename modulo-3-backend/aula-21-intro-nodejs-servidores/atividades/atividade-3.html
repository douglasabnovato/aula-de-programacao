app.js

require('dotenv').config();
const express = require('express');
const rotasTarefas = require('./src/routes/tarefas.routes');
const tratarErro = require('./src/middlewares/erro.middleware');

const app = express();
app.use(express.json());

app.use((req, res, proximo) => {
  console.log(`[${new Date().toISOString()}] ${req.method} ${req.originalUrl}`);
  proximo();
});

app.use('/tarefas', rotasTarefas);
app.use(tratarErro);

const PORTA = process.env.PORT || 3000;
app.listen(PORTA, () => console.log(`API em http://localhost:${PORTA}`));

src/services/tarefas.service.js

let tarefas = [{ id: 1, titulo: 'Estudar camadas', feita: false }];
let proximoId = 2;

const listar = () => tarefas;

const buscarPorId = (id) => tarefas.find(t => t.id === Number(id));

const criar = (titulo) => {
  const nova = { id: proximoId++, titulo, feita: false };
  tarefas.push(nova);
  return nova;
};

module.exports = { listar, buscarPorId, criar };

src/controllers/tarefas.controller.js

const servico = require('../services/tarefas.service');

const listar = (req, res) => res.json(servico.listar());

const detalhar = (req, res) => {
  const tarefa = servico.buscarPorId(req.params.id);
  if (!tarefa) return res.status(404).json({ erro: 'Tarefa não encontrada' });
  res.json(tarefa);
};

const criar = (req, res) => {
  const { titulo } = req.body;
  if (!titulo || !titulo.trim()) {
    return res.status(400).json({ erro: 'O campo titulo é obrigatório' });
  }
  res.status(201).json(servico.criar(titulo.trim()));
};

module.exports = { listar, detalhar, criar };

src/middlewares/erro.middleware.js

// Quatro argumentos: é isso que faz o Express reconhecer como middleware de erro.
module.exports = (erro, req, res, proximo) => {
  console.error(erro);
  res.status(500).json({ erro: 'Erro interno no servidor' });
};

Por que o serviço não conhece req nem res

O serviço recebe e devolve dado puro. Isso permite chamá-lo de um teste, de um script de linha de comando ou de outra rota, sem simular uma requisição HTTP. É a diferença entre código organizado e código só espalhado em mais arquivos.
