Controlador Magro: Antes e Depois

Aula 23 · Módulo 3: Backend e Banco de Dados · ByteClass

Antes — tudo no controlador

const criar = (req, res) => {
  const { titulo, prazo } = req.body;
  if (!titulo || titulo.trim() === '') return res.status(400).json({ erro: 'Título obrigatório' });
  if (prazo && new Date(prazo) < new Date()) return res.status(400).json({ erro: 'Prazo no passado' });

  const existe = tarefas.find(t => t.titulo.toLowerCase() === titulo.trim().toLowerCase());
  if (existe) return res.status(409).json({ erro: 'Já existe tarefa com esse título' });

  const nova = { id: proximoId++, titulo: titulo.trim(), prazo: prazo || null, feita: false };
  tarefas.push(nova);
  res.status(201).json(nova);
};

Nove linhas com quatro responsabilidades: ler HTTP, validar, decidir regra de negócio e gravar dado.

Depois — três arquivos, um motivo cada

// controllers/tarefas.controller.js — só HTTP
const criar = (req, res, proximo) => {
  try {
    const tarefa = servico.criar(req.body);
    res.status(201).json(tarefa);
  } catch (erro) {
    if (erro.codigo === 'VALIDACAO') return res.status(400).json({ erro: erro.message });
    if (erro.codigo === 'DUPLICADO') return res.status(409).json({ erro: erro.message });
    proximo(erro);
  }
};
// services/tarefas.service.js — só regra de negócio
const criar = ({ titulo, prazo }) => {
  if (!titulo || !titulo.trim()) throw erroDe('VALIDACAO', 'Título obrigatório');
  if (prazo && new Date(prazo) < new Date()) throw erroDe('VALIDACAO', 'Prazo no passado');
  if (repositorio.buscarPorTitulo(titulo.trim())) throw erroDe('DUPLICADO', 'Já existe tarefa com esse título');

  return repositorio.salvar({ titulo: titulo.trim(), prazo: prazo || null, feita: false });
};
// repositories/tarefas.repository.js — só dado
let tarefas = [];
let proximoId = 1;

const buscarPorTitulo = (titulo) =>
  tarefas.find(t => t.titulo.toLowerCase() === titulo.toLowerCase());

const salvar = (dados) => {
  const registro = { id: proximoId++, ...dados };
  tarefas.push(registro);
  return registro;
};

O teste que só o "depois" permite

// Sem subir servidor, sem HTTP, sem porta:
const servico = require('./src/services/tarefas.service');

try {
  servico.criar({ titulo: '   ' });
} catch (e) {
  console.log(e.codigo === 'VALIDACAO' ? '✔ recusou título vazio' : '✘ falhou');
}

É esse o ganho concreto. O resto é consequência.
