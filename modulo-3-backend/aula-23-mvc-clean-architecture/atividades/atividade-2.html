Aula 23 · Módulo 3: Backend e Banco de Dados · ByteClass

A regra em uma frase

O que é núcleo não importa o que é borda. A seta aponta sempre para dentro.

core/entities/Tarefa.js

class Tarefa {
  constructor({ id = null, titulo, feita = false }) {
    if (!titulo || !titulo.trim()) {
      throw new Error('Tarefa exige um título');
    }
    this.id = id;
    this.titulo = titulo.trim();
    this.feita = feita;
  }

  concluir() {
    this.feita = true;
    return this;
  }
}

module.exports = Tarefa;

core/usecases/CriarTarefa.js

const Tarefa = require('../entities/Tarefa');

// Recebe o repositorio pelo construtor. Nao sabe se e memoria, arquivo ou Postgres.
class CriarTarefa {
  constructor(repositorio) {
    this.repositorio = repositorio;
  }

  executar({ titulo }) {
    const tarefa = new Tarefa({ titulo });
    return this.repositorio.salvar(tarefa);
  }
}

module.exports = CriarTarefa;

infra/repositories/TarefaMemoriaRepository.js

class TarefaMemoriaRepository {
  constructor() {
    this.itens = [];
    this.proximoId = 1;
  }

  salvar(tarefa) {
    tarefa.id = this.proximoId++;
    this.itens.push(tarefa);
    return tarefa;
  }

  buscarTodos() {
    return this.itens;
  }
}

module.exports = TarefaMemoriaRepository;

A ligação, feita só na borda

const CriarTarefa = require('./core/usecases/CriarTarefa');
const Repositorio = require('./infra/repositories/TarefaMemoriaRepository');

const criarTarefa = new CriarTarefa(new Repositorio());

app.post('/tarefas', (req, res) => {
  const tarefa = criarTarefa.executar(req.body);
  res.status(201).json(tarefa);
});
