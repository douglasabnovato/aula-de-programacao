Testes Unitários com Jest — Código de Referência

Aula 24 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

A entidade e o caso de uso da aula 23 cobertos por dois testes unitários: o caminho feliz e a regra violada. Roda com npm test, sem servidor e sem banco.

src/core/entities/Tarefa.js

// Início: entidade Tarefa com as regras que valem em qualquer ponto do sistema
class Tarefa {
  // cria a tarefa e recusa título vazio
  constructor({ id = null, titulo, feita = false }) {
    if (typeof titulo !== 'string' || titulo.trim() === '') {
      throw new Error('O título da tarefa é obrigatório');
    }
    this.id = id;
    this.titulo = titulo.trim();
    this.feita = feita;
  }
}

module.exports = Tarefa;
// Fim de Tarefa.js

src/core/usecases/CriarTarefa.js

// Início: caso de uso que cria tarefa com qualquer repositório que cumpra o contrato
const Tarefa = require('../entities/Tarefa');

class CriarTarefa {
  // recebe o repositório por injeção de dependência
  constructor(repositorio) {
    this.repositorio = repositorio;
  }

  // valida pela entidade e delega a gravação ao repositório
  async executar({ titulo }) {
    const tarefa = new Tarefa({ titulo });
    return this.repositorio.salvar(tarefa);
  }
}

module.exports = CriarTarefa;
// Fim de CriarTarefa.js

tests/unit/CriarTarefa.test.js

// Início: testes unitários do caso de uso CriarTarefa, sem HTTP e sem banco
const CriarTarefa = require('../../src/core/usecases/CriarTarefa');

// monta um repositório falso que devolve a tarefa com id
function criarRepositorioFalso() {
  return { salvar: jest.fn(async (tarefa) => ({ ...tarefa, id: 1 })) };
}

describe('CriarTarefa', () => {
  // caminho feliz: título válido vira tarefa gravada
  test('cria tarefa com título válido', async () => {
    const repositorio = criarRepositorioFalso();
    const caso = new CriarTarefa(repositorio);

    const resultado = await caso.executar({ titulo: '  Revisar contrato  ' });

    expect(resultado).toEqual({ id: 1, titulo: 'Revisar contrato', feita: false });
    expect(repositorio.salvar).toHaveBeenCalledTimes(1);
  });

  // regra violada: nada chega ao repositório
  test('recusa título vazio e não chama o repositório', async () => {
    const repositorio = criarRepositorioFalso();
    const caso = new CriarTarefa(repositorio);

    await expect(caso.executar({ titulo: '   ' })).rejects.toThrow('O título da tarefa é obrigatório');
    expect(repositorio.salvar).not.toHaveBeenCalled();
  });
});
// Fim de CriarTarefa.test.js

Como rodar

  1. npm install --save-dev jest
  2. Adicione "test": "jest" em scripts no package.json
  3. npm test
  4. npx jest --coverage para ver a cobertura

O que observar

  • O repositório falso é um jest.fn: ele registra cada chamada, e é isso que permite afirmar que salvar não foi chamado.
  • rejects.toThrow só funciona com await na frente — sem ele o teste termina antes da Promise rejeitar.
  • toEqual compara valores, não a classe: a instância de Tarefa espalhada num objeto passa na comparação.
