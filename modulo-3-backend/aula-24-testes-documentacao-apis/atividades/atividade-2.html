Testes de Integração com Supertest — Código de Referência

Aula 24 · Módulo 3: Backend e Banco de Dados · ByteClass

O que esta atividade entrega

O app Express criado por uma fábrica que recebe o repositório, um server.js separado e quatro testes que chamam as rotas de verdade, sem abrir porta.

src/infra/repositories/TarefaMemoriaRepository.js

// Início: repositório em memória que cumpre o contrato usado pelos casos de uso
class TarefaMemoriaRepository {
  // começa com a lista informada ou vazia
  constructor(iniciais = []) {
    this.tarefas = iniciais.map((t, i) => ({ id: i + 1, titulo: t.titulo, feita: Boolean(t.feita) }));
    this.proximoId = this.tarefas.length + 1;
  }

  // devolve uma cópia de todas as tarefas
  async buscarTodos() {
    return [...this.tarefas];
  }

  // devolve a tarefa do id informado ou null
  async buscarPorId(id) {
    return this.tarefas.find((t) => t.id === Number(id)) || null;
  }

  // grava a tarefa e devolve com o id gerado
  async salvar(tarefa) {
    const nova = { id: this.proximoId++, titulo: tarefa.titulo, feita: tarefa.feita };
    this.tarefas.push(nova);
    return nova;
  }
}

module.exports = TarefaMemoriaRepository;
// Fim de TarefaMemoriaRepository.js

src/app.js

// Início: fábrica do app Express, sem listen, para poder ser testado
const express = require('express');
const CriarTarefa = require('./core/usecases/CriarTarefa');

// monta o app usando o repositório recebido
function criarApp(repositorio) {
  const app = express();
  const criarTarefa = new CriarTarefa(repositorio);
  app.use(express.json());

  // lista todas as tarefas
  app.get('/tarefas', async (req, res) => {
    res.json(await repositorio.buscarTodos());
  });

  // detalha uma tarefa ou responde 404
  app.get('/tarefas/:id', async (req, res) => {
    const tarefa = await repositorio.buscarPorId(req.params.id);
    if (!tarefa) return res.status(404).json({ erro: 'Tarefa não encontrada' });
    res.json(tarefa);
  });

  // cria a tarefa pelo caso de uso e traduz a regra violada em 400
  app.post('/tarefas', async (req, res) => {
    try {
      const tarefa = await criarTarefa.executar({ titulo: req.body.titulo });
      res.status(201).json(tarefa);
    } catch (erro) {
      res.status(400).json({ erro: erro.message });
    }
  });

  return app;
}

module.exports = criarApp;
// Fim de app.js

server.js

// Início: ponto de entrada que liga o app a uma porta real
require('dotenv').config();
const criarApp = require('./src/app');
const TarefaMemoriaRepository = require('./src/infra/repositories/TarefaMemoriaRepository');

const app = criarApp(new TarefaMemoriaRepository([{ titulo: 'Estudar testes' }]));
const PORTA = process.env.PORT || 3000;

app.listen(PORTA, () => console.log(`API em http://localhost:${PORTA}`));

tests/integration/tarefas.test.js

// Início: testes de integração das rotas de tarefas com Supertest
const request = require('supertest');
const criarApp = require('../../src/app');
const TarefaMemoriaRepository = require('../../src/infra/repositories/TarefaMemoriaRepository');

describe('API de tarefas', () => {
  let app;

  // recria repositório e app antes de cada teste para isolar o estado
  beforeEach(() => {
    app = criarApp(new TarefaMemoriaRepository([{ titulo: 'Tarefa inicial' }]));
  });

  // listagem devolve array em JSON
  test('GET /tarefas responde 200 com array', async () => {
    const resposta = await request(app).get('/tarefas');
    expect(resposta.status).toBe(200);
    expect(resposta.headers['content-type']).toMatch(/json/);
    expect(Array.isArray(resposta.body)).toBe(true);
  });

  // criação devolve o recurso com id
  test('POST /tarefas cria e responde 201', async () => {
    const resposta = await request(app).post('/tarefas').send({ titulo: 'Ligar para fornecedor' });
    expect(resposta.status).toBe(201);
    expect(resposta.body).toMatchObject({ id: 2, titulo: 'Ligar para fornecedor', feita: false });
  });

  // entrada inválida é recusada
  test('POST /tarefas sem título responde 400', async () => {
    const resposta = await request(app).post('/tarefas').send({});
    expect(resposta.status).toBe(400);
    expect(resposta.body.erro).toBe('O título da tarefa é obrigatório');
  });

  // id inexistente
  test('GET /tarefas/9999 responde 404', async () => {
    const resposta = await request(app).get('/tarefas/9999');
    expect(resposta.status).toBe(404);
  });
});
// Fim de tarefas.test.js

Como rodar

  1. npm install express dotenv
  2. npm install --save-dev jest supertest
  3. npm test
  4. node server.js para usar a API normalmente

O que observar

  • request(app) recebe o app, não uma URL: o Supertest abre uma porta temporária sozinho e fecha ao terminar.
  • O beforeEach cria um repositório novo a cada teste; sem ele, o id esperado no teste de criação mudaria conforme a ordem de execução.
  • A fábrica criarApp(repositorio) é a mesma injeção de dependência da aula 23, agora usada para testar.
