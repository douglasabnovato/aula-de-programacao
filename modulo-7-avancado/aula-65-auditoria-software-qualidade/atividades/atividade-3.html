Script de Hotspots a partir do Histórico do Git — Código de Referência

Aula 65 · Módulo 7: Avançado (Engenharia e Arquitetura Master) · ByteClass

O que esta atividade entrega

Um script Node que lê o histórico do Git, conta quantas vezes cada arquivo mudou, cruza com o tamanho atual e imprime o ranking de hotspots. É o ponto de partida do relatório de auditoria.

scripts/hotspots.js

// Início: ranking de hotspots cruzando frequência de mudança e tamanho de arquivo
const { execSync } = require('child_process');
const fs = require('fs');

const PERIODO = process.argv[2] || '12.months';
const IGNORAR = [/package-lock\.json$/, /yarn\.lock$/, /^dist\//, /^build\//, /\.min\.js$/, /\.md$/];

// Lê o histórico e devolve um mapa arquivo -> número de commits que o alteraram
function contarMudancas(periodo) {
  const saida = execSync(`git log --since=${periodo} --numstat --format=`, { encoding: 'utf8', maxBuffer: 50 * 1024 * 1024 });
  const mudancas = new Map();
  for (const linha of saida.split('\n')) {
    const partes = linha.split('\t');
    if (partes.length !== 3) continue;
    const arquivo = partes[2];
    if (IGNORAR.some((padrao) => padrao.test(arquivo))) continue;
    mudancas.set(arquivo, (mudancas.get(arquivo) || 0) + 1);
  }
  return mudancas;
}

// Conta as linhas atuais do arquivo, ou zero se ele não existe mais
function contarLinhas(arquivo) {
  if (!fs.existsSync(arquivo)) return 0;
  return fs.readFileSync(arquivo, 'utf8').split('\n').length;
}

// Monta o ranking com pontuação normalizada entre 0 e 1
function calcularHotspots(mudancas) {
  const itens = [...mudancas]
    .map(([arquivo, commits]) => ({ arquivo, commits, linhas: contarLinhas(arquivo) }))
    .filter((item) => item.linhas > 0);
  const maxCommits = Math.max(...itens.map((i) => i.commits));
  const maxLinhas = Math.max(...itens.map((i) => i.linhas));
  return itens
    .map((i) => ({ ...i, pontuacao: +((i.commits / maxCommits) * (i.linhas / maxLinhas)).toFixed(3) }))
    .sort((a, b) => b.pontuacao - a.pontuacao);
}

// Imprime os dez primeiros em formato de tabela
function main() {
  const ranking = calcularHotspots(contarMudancas(PERIODO)).slice(0, 10);
  console.table(ranking);
}

main();
// Fim de scripts/hotspots.js

Como rodar

  1. Entre na raiz de um repositório Git com histórico
  2. node scripts/hotspots.js 12.months
  3. Rode com 6.months e compare se o ranking muda
  4. Para os cinco primeiros, rode npx eslint --rule 'complexity: [warn, 10]' caminho/do/arquivo.js

O que observar

  • Arquivo grande que ninguém altera fica com pontuação baixa: não é ali que a dívida cobra juros.
  • Sem a lista IGNORAR, o lockfile domina o ranking, o que mostra por que todo dado precisa de limpeza antes da análise.
  • O número de linhas é uma aproximação grosseira; o cruzamento com a complexidade do lint confirma ou descarta cada candidato.
