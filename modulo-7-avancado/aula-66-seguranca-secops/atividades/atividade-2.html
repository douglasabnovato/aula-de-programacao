O que esta atividade entrega

As duas versões das rotas da Nota Certa, lado a lado, e os testes que provam o ataque e depois a correção. Use apenas no seu ambiente local de laboratório.

src/notas.vulneravel.js

// Início: versão VULNERÁVEL, usada só no laboratório para demonstrar IDOR e injeção
const express = require('express');
const { pool } = require('./db');
const { autenticar } = require('./auth');

const router = express.Router();

router.get('/notas/:id', autenticar, async (req, res) => {
  const { rows } = await pool.query(`SELECT * FROM notas WHERE id = ${req.params.id}`);
  if (!rows[0]) return res.status(404).json({ erro: 'Nota não encontrada' });
  res.json(rows[0]);
});

router.get('/notas', autenticar, async (req, res) => {
  const termo = req.query.cliente || '';
  const { rows } = await pool.query(`SELECT * FROM notas WHERE empresa_id = ${req.usuario.empresaId} AND cliente = '${termo}'`);
  res.json(rows);
});

module.exports = router;
// Fim de src/notas.vulneravel.js

src/notas.routes.js

// Início: versão CORRIGIDA, com autorização por dono e consultas parametrizadas
const express = require('express');
const { pool } = require('./db');
const { autenticar } = require('./auth');

const router = express.Router();

// Busca uma nota garantindo que ela pertence à empresa do usuário logado
async function buscarNotaDaEmpresa(id, empresaId) {
  const { rows } = await pool.query(
    'SELECT id, numero, cliente, valor, emitida_em FROM notas WHERE id = $1 AND empresa_id = $2',
    [id, empresaId]
  );
  return rows[0];
}

router.get('/notas/:id', autenticar, async (req, res) => {
  const id = Number.parseInt(req.params.id, 10);
  if (!Number.isInteger(id)) return res.status(400).json({ erro: 'Identificador inválido' });
  const nota = await buscarNotaDaEmpresa(id, req.usuario.empresaId);
  if (!nota) return res.status(404).json({ erro: 'Nota não encontrada' });
  res.json(nota);
});

router.get('/notas', autenticar, async (req, res) => {
  const termo = String(req.query.cliente || '');
  const { rows } = await pool.query(
    'SELECT id, numero, cliente, valor, emitida_em FROM notas WHERE empresa_id = $1 AND cliente ILIKE $2 ORDER BY emitida_em DESC LIMIT 50',
    [req.usuario.empresaId, `%${termo}%`]
  );
  res.json(rows);
});

module.exports = router;
// Fim de src/notas.routes.js

tests/seguranca.test.js

// Início: testes de segurança que reproduzem os ataques contra a rota corrigida
const request = require('supertest');
const app = require('../src/app');
const { tokenDe } = require('./helpers');

describe('Nota Certa — ataques de laboratório', () => {
  test('IDOR: empresa A não lê nota da empresa B', async () => {
    const res = await request(app).get('/notas/4').set('Authorization', `Bearer ${tokenDe('empresaA')}`);
    expect(res.status).toBe(404);
  });

  test('Injeção: termo malicioso não amplia o resultado', async () => {
    const res = await request(app)
      .get('/notas')
      .query({ cliente: "' OR '1'='1" })
      .set('Authorization', `Bearer ${tokenDe('empresaA')}`);
    expect(res.status).toBe(200);
    expect(res.body).toHaveLength(0);
  });
});
// Fim de tests/seguranca.test.js

Como rodar

  1. npm install express pg jsonwebtoken e npm install -D jest supertest
  2. Suba o PostgreSQL local e rode o script de sementes com duas empresas
  3. Monte o app com notas.vulneravel.js e rode npx jest: os dois testes falham, provando os ataques
  4. Troque para notas.routes.js e rode de novo: os dois passam

O que observar

  • A correção do IDOR responde 404, e não 403, para não confirmar ao atacante que a nota existe.
  • A consulta parametrizada envia o termo separado do comando: o banco nunca interpreta o texto do usuário como SQL.
  • Os testes ficam na suíte para sempre; se alguém reintroduzir a falha, o pipeline acusa.
