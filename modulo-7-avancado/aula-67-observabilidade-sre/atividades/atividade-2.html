Instrumentação com OpenTelemetry — Código de Referência

Aula 67 · Módulo 7: Avançado (Engenharia e Arquitetura Master) · ByteClass

O que esta atividade entrega

A instrumentação de dois serviços Node com o SDK do OpenTelemetry, exportando via OTLP para o Jaeger, e um span manual com atributo e registro de erro. Os nomes dos pacotes seguem a documentação oficial; confira as versões atuais antes de instalar.

instrumentation.js

// Início: inicialização do OpenTelemetry, carregada antes de qualquer outro módulo
const { NodeSDK } = require('@opentelemetry/sdk-node');
const { getNodeAutoInstrumentations } = require('@opentelemetry/auto-instrumentations-node');
const { OTLPTraceExporter } = require('@opentelemetry/exporter-trace-otlp-http');

const sdk = new NodeSDK({
  serviceName: process.env.OTEL_SERVICE_NAME || 'servico-sem-nome',
  traceExporter: new OTLPTraceExporter({
    url: process.env.OTEL_EXPORTER_OTLP_TRACES_ENDPOINT || 'http://localhost:4318/v1/traces',
  }),
  instrumentations: [getNodeAutoInstrumentations()],
});

sdk.start();

// Envia os spans pendentes antes de o processo terminar
async function encerrar() {
  await sdk.shutdown();
  process.exit(0);
}

process.on('SIGTERM', encerrar);
// Fim de instrumentation.js

pedidos/app.js

// Início: serviço de pedidos que consulta o estoque e cria um span manual
const express = require('express');
const { trace, SpanStatusCode } = require('@opentelemetry/api');

const app = express();
app.use(express.json());
const tracer = trace.getTracer('pedidos');
const ESTOQUE_URL = process.env.ESTOQUE_URL || 'http://localhost:3002';

// Calcula o desconto dentro de um span próprio com o valor do carrinho como atributo
function calcularDesconto(itens) {
  return tracer.startActiveSpan('calcular-desconto', (span) => {
    const total = itens.reduce((soma, item) => soma + item.preco * item.quantidade, 0);
    span.setAttribute('carrinho.valor_total', total);
    const desconto = total > 200 ? total * 0.05 : 0;
    span.end();
    return { total, desconto };
  });
}

app.post('/pedidos', async (req, res) => {
  const itens = req.body.itens || [];
  const span = trace.getActiveSpan();
  for (const item of itens) {
    const resposta = await fetch(`${ESTOQUE_URL}/estoque/${encodeURIComponent(item.sku)}`);
    if (resposta.status === 409) {
      const erro = new Error(`Sem estoque para ${item.sku}`);
      span?.recordException(erro);
      span?.setStatus({ code: SpanStatusCode.ERROR, message: erro.message });
      return res.status(409).json({ erro: erro.message });
    }
  }
  res.status(201).json(calcularDesconto(itens));
});

app.listen(3001, () => console.log('pedidos na porta 3001'));
// Fim de pedidos/app.js

docker-compose.yml

# Início: Jaeger recebendo OTLP e exibindo a interface de consulta
services:
  jaeger:
    image: jaegertracing/all-in-one:latest
    ports:
      - "16686:16686"
      - "4318:4318"
    environment:
      COLLECTOR_OTLP_ENABLED: "true"
# Fim de docker-compose.yml

Como rodar

  1. docker compose up -d e abra http://localhost:16686
  2. Em cada serviço: npm install express @opentelemetry/api @opentelemetry/sdk-node @opentelemetry/auto-instrumentations-node @opentelemetry/exporter-trace-otlp-http
  3. OTEL_SERVICE_NAME=pedidos node --require ./instrumentation.js pedidos/app.js e o equivalente para o estoque na porta 3002
  4. Faça um POST /pedidos e procure o trace no Jaeger pelo serviço pedidos

O que observar

  • O cabeçalho traceparent é injetado automaticamente no fetch; é ele que liga o span do estoque ao do pedido.
  • Se a instrumentação for carregada depois do Express, os módulos já foram importados sem os ganchos e as rotas não geram spans.
  • O span manual calcular-desconto aparece como filho da requisição, com o atributo de valor pesquisável no Jaeger.
