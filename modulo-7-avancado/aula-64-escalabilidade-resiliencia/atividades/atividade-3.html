Health Checks, Bulkhead e Desligamento Gracioso — Código de Referência

Aula 64 · Módulo 7: Avançado (Engenharia e Arquitetura Master) · ByteClass

O que esta atividade entrega

Uma API Express com endpoints de vida e prontidão, um limitador de concorrência por rota, pools de conexão separados e desligamento gracioso no SIGTERM. É a base para rodar várias réplicas atrás de um balanceador sem perder requisições.

src/bulkhead.js

// Início: middleware que limita execuções simultâneas de uma rota (compartimento)

// Cria um middleware com limite fixo de requisições em andamento
function bulkhead({ maxConcorrentes = 2, retryAfterSeg = 5 } = {}) {
  let emAndamento = 0;
  return function (req, res, next) {
    if (emAndamento >= maxConcorrentes) {
      res.set('Retry-After', String(retryAfterSeg));
      return res.status(503).json({ erro: 'Capacidade da rota esgotada, tente novamente em instantes' });
    }
    emAndamento += 1;
    res.on('close', () => { emAndamento -= 1; });
    next();
  };
}

module.exports = { bulkhead };
// Fim de src/bulkhead.js

src/server.js

// Início: API de agendamento com health checks, bulkhead e desligamento gracioso
const express = require('express');
const { Pool } = require('pg');
const { bulkhead } = require('./bulkhead');

const app = express();
const poolPrincipal = new Pool({ connectionString: process.env.DATABASE_URL, max: 8 });
const poolRelatorios = new Pool({ connectionString: process.env.DATABASE_URL, max: 2 });
let encerrando = false;

// Testa o banco com prazo máximo para o health check não travar
async function bancoRespondendo(pool, prazoMs = 500) {
  const prazo = new Promise((_, rejeitar) => setTimeout(() => rejeitar(new Error('prazo esgotado')), prazoMs));
  await Promise.race([pool.query('SELECT 1'), prazo]);
}

app.get('/health/live', (req, res) => res.status(200).json({ status: 'vivo' }));

app.get('/health/ready', async (req, res) => {
  if (encerrando) return res.status(503).json({ status: 'encerrando' });
  try {
    await bancoRespondendo(poolPrincipal);
    res.status(200).json({ status: 'pronto' });
  } catch (erro) {
    res.status(503).json({ status: 'indisponível', motivo: erro.message });
  }
});

app.get('/agendamentos', async (req, res, next) => {
  try {
    const { rows } = await poolPrincipal.query('SELECT id, paciente, horario FROM agendamentos ORDER BY horario LIMIT 50');
    res.json({ dados: rows });
  } catch (erro) {
    next(erro);
  }
});

app.get('/relatorios/mensal', bulkhead({ maxConcorrentes: 2 }), async (req, res, next) => {
  try {
    const { rows } = await poolRelatorios.query(
      "SELECT date_trunc('day', horario) AS dia, count(*) AS total FROM agendamentos GROUP BY 1 ORDER BY 1"
    );
    res.json({ dados: rows });
  } catch (erro) {
    next(erro);
  }
});

app.use((erro, req, res, next) => {
  console.error(erro);
  res.status(500).json({ erro: 'Falha interna' });
});

const servidor = app.listen(process.env.PORT || 3000, () => console.log('API no ar'));

// Para de receber tráfego, termina o que está em andamento e fecha as conexões
function desligar() {
  encerrando = true;
  setTimeout(() => {
    servidor.close(async () => {
      await Promise.all([poolPrincipal.end(), poolRelatorios.end()]);
      process.exit(0);
    });
  }, Number(process.env.ATRASO_SAIDA_MS || 5000));
  setTimeout(() => process.exit(1), 30000).unref();
}

process.on('SIGTERM', desligar);
process.on('SIGINT', desligar);
// Fim de src/server.js

Como rodar

  1. npm install express pg
  2. Defina DATABASE_URL e crie a tabela agendamentos
  3. node src/server.js
  4. Teste /health/live e /health/ready, depois pare o banco e teste de novo
  5. Com o k6, dispare dez chamadas simultâneas a /relatorios/mensal e observe as respostas 503

O que observar

  • O endpoint de vida não consulta o banco: se consultasse, uma lentidão do banco faria o orquestrador reiniciar réplicas saudáveis.
  • O atraso antes de fechar o servidor dá tempo ao balanceador de ler o /ready como 503 e parar de mandar tráfego.
  • O pool de relatórios com duas conexões garante que o agendamento sempre tenha as suas oito livres.
