Timeout, Retry com Jitter e Circuit Breaker — Código de Referência

Aula 64 · Módulo 7: Avançado (Engenharia e Arquitetura Master) · ByteClass

O que esta atividade entrega

Um módulo que envolve chamadas HTTP com timeout, retry exponencial com jitter e disjuntor, e um exemplo de uso que devolve frete estimado quando o fornecedor está fora do ar. Requer Node 18 ou superior (fetch nativo).

resiliencia.js

// Início: utilitários de resiliência para chamadas HTTP a serviços externos
const ESTADOS = { FECHADO: 'fechado', ABERTO: 'aberto', SEMIABERTO: 'semiaberto' };

// Decide se vale a pena tentar de novo com base no tipo de falha
function ehTransitorio(erro) {
  if (erro.name === 'TimeoutError' || erro.name === 'AbortError') return true;
  if (erro.status) return [429, 502, 503, 504].includes(erro.status);
  return erro.code === 'ECONNREFUSED' || erro.code === 'ECONNRESET';
}

// Espera o tempo indicado em milissegundos
function esperar(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

// Faz a requisição com limite de tempo e transforma status de erro em exceção
async function fetchComTimeout(url, { timeoutMs = 2000, ...opcoes } = {}) {
  const res = await fetch(url, { ...opcoes, signal: AbortSignal.timeout(timeoutMs) });
  if (!res.ok) {
    const erro = new Error(`HTTP ${res.status}`);
    erro.status = res.status;
    throw erro;
  }
  return res.json();
}

// Repete a operação com backoff exponencial e jitter completo
async function comRetry(operacao, { tentativas = 3, baseMs = 200, maxMs = 3000 } = {}) {
  let ultimoErro;
  for (let i = 0; i < tentativas; i++) {
    try {
      return await operacao();
    } catch (erro) {
      ultimoErro = erro;
      if (!ehTransitorio(erro) || i === tentativas - 1) break;
      const teto = Math.min(maxMs, baseMs * 2 ** i);
      await esperar(Math.random() * teto);
    }
  }
  throw ultimoErro;
}

class CircuitBreaker {
  // Guarda os limites e começa no estado fechado
  constructor({ limiteFalhas = 5, reaberturaMs = 10000 } = {}) {
    this.limiteFalhas = limiteFalhas;
    this.reaberturaMs = reaberturaMs;
    this.estado = ESTADOS.FECHADO;
    this.falhas = 0;
    this.abertoEm = 0;
  }

  // Executa a operação respeitando o estado atual do disjuntor
  async executar(operacao, alternativa) {
    if (this.estado === ESTADOS.ABERTO) {
      if (Date.now() - this.abertoEm < this.reaberturaMs) return alternativa();
      this.estado = ESTADOS.SEMIABERTO;
    }
    try {
      const resultado = await operacao();
      this.registrarSucesso();
      return resultado;
    } catch (erro) {
      this.registrarFalha();
      return alternativa(erro);
    }
  }

  // Zera o contador e fecha o circuito depois de um sucesso
  registrarSucesso() {
    this.falhas = 0;
    this.estado = ESTADOS.FECHADO;
  }

  // Conta a falha e abre o circuito ao atingir o limite ou ao falhar no teste semiaberto
  registrarFalha() {
    this.falhas += 1;
    if (this.estado === ESTADOS.SEMIABERTO || this.falhas >= this.limiteFalhas) {
      this.estado = ESTADOS.ABERTO;
      this.abertoEm = Date.now();
    }
  }
}

module.exports = { fetchComTimeout, comRetry, CircuitBreaker, ehTransitorio };
// Fim de resiliencia.js

frete.service.js

// Início: serviço de frete da Casa Ipê protegido por timeout, retry e disjuntor
const { fetchComTimeout, comRetry, CircuitBreaker } = require('./resiliencia');

const FRETE_URL = process.env.FRETE_URL || 'http://localhost:4000/frete';
const disjuntor = new CircuitBreaker({ limiteFalhas: 5, reaberturaMs: 10000 });

// Valor padrão usado quando o fornecedor não responde
function freteEstimado(cep) {
  return { cep, valor: 29.9, prazoDias: 7, estimado: true };
}

// Consulta o frete real e cai para a estimativa se o fornecedor falhar
async function calcularFrete(cep) {
  return disjuntor.executar(
    () => comRetry(() => fetchComTimeout(`${FRETE_URL}?cep=${encodeURIComponent(cep)}`, { timeoutMs: 2000 })),
    () => freteEstimado(cep)
  );
}

module.exports = { calcularFrete, disjuntor };
// Fim de frete.service.js

Como rodar

  1. Suba um servidor falso de frete na porta 4000 que atrasa e falha aleatoriamente
  2. Chame calcularFrete('36000-000') em um laço de 50 iterações e imprima disjuntor.estado a cada volta
  3. Desligue o servidor falso no meio do laço e observe a troca para aberto
  4. Religue e espere 10 segundos para ver o estado semiaberto voltar a fechado

O que observar

  • O retry fica dentro do disjuntor: uma chamada com três tentativas conta como uma falha só.
  • Status 400 e 404 não são repetidos, porque tentar de novo não muda uma requisição errada.
  • No estado aberto a resposta é imediata e marcada como estimado: true, o que permite à interface avisar o cliente.
