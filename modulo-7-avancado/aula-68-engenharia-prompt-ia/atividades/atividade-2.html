Assistente com Uso de Ferramentas — Código de Referência

Aula 68 · Módulo 7: Avançado (Engenharia e Arquitetura Master) · ByteClass

O que esta atividade entrega

Um assistente de pedidos que declara uma ferramenta, executa a chamada pedida pelo modelo no próprio código, devolve o resultado e repete até a resposta final, com limite de iterações e autorização feita pela aplicação.

src/assistente.js

// Início: assistente de pedidos da Estante Nova usando tool use da Messages API
const API_URL = 'https://api.anthropic.com/v1/messages';
const MAX_ITERACOES = 5;

const PEDIDOS = {
  '100231': { cliente: 'c-01', status: 'em transporte', previsao: '2026-10-02' },
  '100232': { cliente: 'c-02', status: 'entregue', previsao: '2026-09-20' },
};

const FERRAMENTAS = [
  {
    name: 'consultar_pedido',
    description: 'Consulta o status e a previsão de entrega de um pedido do cliente logado. Use quando o cliente perguntar sobre um pedido específico.',
    input_schema: {
      type: 'object',
      properties: { numero: { type: 'string', pattern: '^[0-9]{6}$', description: 'Número do pedido com seis dígitos' } },
      required: ['numero'],
    },
  },
];

// Executa a ferramenta no código da aplicação, usando o cliente da sessão e não o do modelo
function executarFerramenta(nome, entrada, clienteLogado) {
  if (nome !== 'consultar_pedido') return { erro: true, conteudo: 'Ferramenta desconhecida' };
  if (!/^[0-9]{6}$/.test(entrada.numero || '')) return { erro: true, conteudo: 'Número de pedido inválido' };
  const pedido = PEDIDOS[entrada.numero];
  if (!pedido || pedido.cliente !== clienteLogado) return { erro: true, conteudo: 'Pedido não encontrado para este cliente' };
  return { erro: false, conteudo: JSON.stringify({ status: pedido.status, previsao: pedido.previsao }) };
}

// Envia a conversa para a API e devolve o corpo da resposta
async function chamarModelo(mensagens) {
  const resposta = await fetch(API_URL, {
    method: 'POST',
    headers: {
      'content-type': 'application/json',
      'x-api-key': process.env.ANTHROPIC_API_KEY,
      'anthropic-version': '2023-06-01',
    },
    body: JSON.stringify({
      model: process.env.ANTHROPIC_MODEL,
      max_tokens: 400,
      system: 'Você é o assistente de pedidos da Estante Nova. Responda em português, de forma curta, usando apenas dados vindos das ferramentas.',
      tools: FERRAMENTAS,
      messages: mensagens,
    }),
    signal: AbortSignal.timeout(20000),
  });
  if (!resposta.ok) throw new Error(`API respondeu ${resposta.status}`);
  return resposta.json();
}

// Conduz a conversa até o modelo parar de pedir ferramentas ou atingir o limite
async function atender(pergunta, clienteLogado) {
  const mensagens = [{ role: 'user', content: pergunta }];
  for (let i = 0; i < MAX_ITERACOES; i++) {
    const corpo = await chamarModelo(mensagens);
    mensagens.push({ role: 'assistant', content: corpo.content });
    if (corpo.stop_reason !== 'tool_use') {
      return corpo.content.filter((b) => b.type === 'text').map((b) => b.text).join('');
    }
    const resultados = corpo.content
      .filter((b) => b.type === 'tool_use')
      .map((b) => {
        const r = executarFerramenta(b.name, b.input, clienteLogado);
        return { type: 'tool_result', tool_use_id: b.id, content: r.conteudo, is_error: r.erro };
      });
    mensagens.push({ role: 'user', content: resultados });
  }
  return 'Não consegui concluir o atendimento agora. Um atendente vai continuar a conversa.';
}

module.exports = { atender, executarFerramenta };
// Fim de src/assistente.js

Como rodar

  1. Use o mesmo .env da atividade anterior
  2. node --env-file=.env -e "require('./src/assistente').atender('Onde está o pedido 100231?', 'c-01').then(console.log)"
  3. Repita com o pedido 100232 ainda como c-01 e observe a resposta de não encontrado
  4. Imprima o array mensagens ao final para ver cada rodada de tool_use e tool_result

O que observar

  • O modelo nunca acessa os dados: ele só pede a chamada, e quem decide executá-la é executarFerramenta.
  • O cliente vem da sessão da aplicação; mesmo que alguém convença o modelo a pedir outro número, o pedido de outra pessoa não é devolvido.
  • O limite de iterações impede um laço infinito de chamadas, que custaria tokens sem entregar resposta.
