Suíte de Avaliação do Classificador — Código de Referência

Aula 68 · Módulo 7: Avançado (Engenharia e Arquitetura Master) · ByteClass

O que esta atividade entrega

Um conjunto de casos com resposta esperada (incluindo tentativas de injeção) e um script que roda a avaliação com concorrência limitada, mostra acerto por categoria e falha abaixo do limite, para uso antes de publicar mudança de prompt.

evals/avaliar.js

// Início: executa a suíte de casos contra o classificador e falha abaixo do limite
const casos = require('./casos.json');
const { classificar } = require('../src/classificador');

const LIMITE = Number(process.env.EVAL_LIMITE || 0.9);
const CONCORRENCIA = 3;

// Executa uma lista de tarefas com no máximo N simultâneas
async function emLotes(itens, n, tarefa) {
  const resultados = [];
  for (let i = 0; i < itens.length; i += n) {
    const lote = itens.slice(i, i + n);
    resultados.push(...(await Promise.all(lote.map(tarefa))));
  }
  return resultados;
}

// Roda cada caso e registra se a categoria bateu com a esperada
async function avaliarCaso(caso) {
  const { categoria } = await classificar(caso.mensagem);
  return { ...caso, obtido: categoria, acertou: categoria === caso.esperado };
}

// Agrupa acertos por categoria esperada
function porCategoria(resultados) {
  const grupos = {};
  for (const r of resultados) {
    grupos[r.esperado] ??= { total: 0, acertos: 0 };
    grupos[r.esperado].total += 1;
    if (r.acertou) grupos[r.esperado].acertos += 1;
  }
  return grupos;
}

// Imprime o relatório e define o código de saída
async function main() {
  const resultados = await emLotes(casos, CONCORRENCIA, avaliarCaso);
  const acerto = resultados.filter((r) => r.acertou).length / resultados.length;
  console.table(porCategoria(resultados));
  console.log('Erros:', resultados.filter((r) => !r.acertou).map(({ mensagem, esperado, obtido }) => ({ mensagem, esperado, obtido })));
  console.log(`Acerto geral: ${(acerto * 100).toFixed(1)}% (limite ${(LIMITE * 100).toFixed(0)}%) com modelo ${process.env.ANTHROPIC_MODEL}`);
  process.exitCode = acerto >= LIMITE ? 0 : 1;
}

main();
// Fim de evals/avaliar.js

Como rodar

  1. Complete casos.json até 30 casos comuns e 5 adversariais
  2. Adicione ao package.json o script "eval": "node --env-file=.env evals/avaliar.js"
  3. npm run eval e anote o acerto geral
  4. Troque o prompt pela versão ingênua, rode de novo e compare

O que observar

  • Os casos adversariais têm resposta esperada normal: a defesa funcionou quando a injeção não muda a categoria.
  • O último caso tenta fechar a tag de dado; por isso a validação da saída continua necessária mesmo com o prompt reforçado.
  • O relatório registra o modelo usado, porque trocar de modelo exige rodar a suíte de novo tanto quanto mudar o prompt.
