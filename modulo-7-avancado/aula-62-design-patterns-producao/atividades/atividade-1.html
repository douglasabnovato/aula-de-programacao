Factory: o problema que ela resolve

// ANTES — a decisão de qual usar está espalhada pelo código
function enviarNotificacao(tipo, destino, mensagem) {
  if (tipo === 'email')    { return new ServicoEmail().enviar(destino, mensagem); }
  if (tipo === 'sms')      { return new ServicoSms().enviar(destino, mensagem); }
  if (tipo === 'whatsapp') { return new ServicoWhats().enviar(destino, mensagem); }
  throw new Error('tipo desconhecido');
}

Esse bloco se repete em cada lugar que notifica. Acrescentar push exige achar todos.

// DEPOIS — uma fábrica, um lugar para mudar
const FABRICAS = {
  email:    () => new ServicoEmail(),
  sms:      () => new ServicoSms(),
  whatsapp: () => new ServicoWhats()
};

function criarNotificador(tipo) {
  const fabrica = FABRICAS[tipo];
  if (!fabrica) throw new Error(`Notificador desconhecido: ${tipo}`);
  return fabrica();
}

// Uso — quem chama não sabe qual implementação veio:
criarNotificador(config.canal).enviar(destino, mensagem);

Builder: quando o construtor tem parâmetros demais

// ANTES — o que é o terceiro true?
const relatorio = new Relatorio('vendas', '2026-09', true, false, true, 'pdf', 30);
// DEPOIS
class ConstrutorDeRelatorio {
  constructor() { this.config = { formato: 'pdf', diasDeRetencao: 30 }; }

  doTipo(tipo)            { this.config.tipo = tipo; return this; }
  noPeriodo(periodo)      { this.config.periodo = periodo; return this; }
  comGraficos()           { this.config.graficos = true; return this; }
  comDetalhamento()       { this.config.detalhado = true; return this; }
  noFormato(f)            { this.config.formato = f; return this; }

  construir() {
    if (!this.config.tipo)    throw new Error('Relatório exige tipo');
    if (!this.config.periodo) throw new Error('Relatório exige período');
    return new Relatorio(this.config);
  }
}

// Uso — legível sem consultar a assinatura:
const relatorio = new ConstrutorDeRelatorio()
  .doTipo('vendas')
  .noPeriodo('2026-09')
  .comGraficos()
  .construir();

Singleton: em JavaScript, o módulo já é um

// Não precisa de classe com getInstance: o módulo é avaliado
// uma vez e o mesmo objeto é devolvido em todo require/import.
let conexao = null;

function obterConexao() {
  if (!conexao) conexao = criarConexao(process.env.DATABASE_URL);
  return conexao;
}

module.exports = { obterConexao };

O Singleton é criticado com razão: ele introduz estado global e dificulta teste, porque não dá para substituir por um dublê. Quando puder, prefira injetar a dependência em vez de buscá-la de um lugar único.
